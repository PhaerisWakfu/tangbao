function escapeHtml(value) {
  return value
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;");
}

function inlineMarkdown(value) {
  let text = escapeHtml(value);
  text = text.replace(/`([^`\n]+)`/g, "<code>$1</code>");
  text = text.replace(/\[([^\]]+)\]\((https?:\/\/[^\s)]+)\)/g, '<a href="$2" target="_blank" rel="noopener noreferrer">$1</a>');
  text = text.replace(/\*\*([^*]+)\*\*/g, "<strong>$1</strong>");
  text = text.replace(/(^|[^*])\*([^*\n]+)\*/g, "$1<em>$2</em>");
  text = text.replace(/~~([^~]+)~~/g, "<del>$1</del>");
  return text.replace(/\n/g, "<br>");
}

function isTableStart(lines, index) {
  return /^\s*\|/.test(lines[index] || "") && /^\s*\|?\s*:?-{3,}/.test(lines[index + 1] || "");
}

function splitRow(line) {
  return line.trim().replace(/^\|/, "").replace(/\|$/, "").split("|").map((cell) => cell.trim());
}

function renderMarkdown(source) {
  const lines = String(source || "").replace(/\r\n/g, "\n").replace(/\r/g, "\n").split("\n");
  const blocks = [];
  let index = 0;
  while (index < lines.length) {
    const line = lines[index];
    if (line.trim().startsWith("```")) {
      const code = [];
      index += 1;
      while (index < lines.length && !lines[index].trim().startsWith("```")) {
        code.push(lines[index]);
        index += 1;
      }
      if (index < lines.length) index += 1;
      blocks.push(`<pre><code>${escapeHtml(code.join("\n"))}</code></pre>`);
      continue;
    }
    if (/^\s*\$\$/.test(line)) {
      const block = line.trim();
      if (block.slice(2).includes("$$")) {
        blocks.push(`<div class="math-block">${escapeHtml(block)}</div>`);
        index += 1;
        continue;
      }
      const math = [block];
      index += 1;
      while (index < lines.length && !lines[index].includes("$$")) {
        math.push(lines[index]);
        index += 1;
      }
      if (index < lines.length) {
        math.push(lines[index].trim());
        index += 1;
      }
      blocks.push(`<div class="math-block">${escapeHtml(math.join("\n"))}</div>`);
      continue;
    }
    if (isTableStart(lines, index)) {
      const head = splitRow(lines[index]);
      index += 2;
      const body = [];
      while (index < lines.length && /^\s*\|/.test(lines[index])) {
        body.push(splitRow(lines[index]));
        index += 1;
      }
      const cells = (row, tag) => row.map((cell) => `<${tag}>${inlineMarkdown(cell)}</${tag}>`).join("");
      blocks.push(
        `<table><thead><tr>${cells(head, "th")}</tr></thead><tbody>${body.map((row) => `<tr>${cells(row, "td")}</tr>`).join("")}</tbody></table>`,
      );
      continue;
    }
    const heading = /^(#{1,3})\s+(.+)$/.exec(line);
    if (heading) {
      const level = heading[1].length;
      blocks.push(`<h${level}>${inlineMarkdown(heading[2])}</h${level}>`);
      index += 1;
      continue;
    }
    if (/^\s*([-*_])\1{2,}\s*$/.test(line)) {
      blocks.push("<hr>");
      index += 1;
      continue;
    }
    if (/^\s*>\s?/.test(line)) {
      const quote = [];
      while (index < lines.length && /^\s*>\s?/.test(lines[index])) {
        quote.push(lines[index].replace(/^\s*>\s?/, ""));
        index += 1;
      }
      blocks.push(`<blockquote><p>${inlineMarkdown(quote.join("\n"))}</p></blockquote>`);
      continue;
    }
    if (/^\s*([-*+]|\d+\.)\s+/.test(line)) {
      const ordered = /^\s*\d+\.\s+/.test(line);
      const firstNumber = Number((/^\s*(\d+)\./.exec(line) || [])[1] || 1);
      const items = [];
      while (index < lines.length && /^\s*([-*+]|\d+\.)\s+/.test(lines[index])) {
        items.push(lines[index].replace(/^\s*([-*+]|\d+\.)\s+/, ""));
        index += 1;
      }
      const tag = ordered ? "ol" : "ul";
      const start = ordered && firstNumber !== 1 ? ` start="${firstNumber}"` : "";
      blocks.push(`<${tag}${start}>${items.map((item) => `<li>${inlineMarkdown(item)}</li>`).join("")}</${tag}>`);
      continue;
    }
    if (!line.trim()) {
      index += 1;
      continue;
    }
    const para = [line];
    index += 1;
    while (index < lines.length && lines[index].trim() && !/^(#{1,3}\s|>|```|\s*([-*+]|\d+\.)\s|\s*\|)/.test(lines[index]) && !/^\s*([-*_])\1{2,}\s*$/.test(lines[index])) {
      para.push(lines[index]);
      index += 1;
    }
    blocks.push(`<p>${inlineMarkdown(para.join("\n"))}</p>`);
  }
  return blocks.join("");
}

function typesetMath(root) {
  if (!root || typeof renderMathInElement !== "function") return;
  try {
    renderMathInElement(root, {
      delimiters: [
        { left: "$$", right: "$$", display: true },
        { left: "$", right: "$", display: false },
        { left: "\\[", right: "\\]", display: true },
        { left: "\\(", right: "\\)", display: false },
      ],
      throwOnError: false,
      trust: false,
      strict: "ignore",
      ignoredTags: ["script", "noscript", "style", "textarea", "pre", "code", "option"],
      ignoredClasses: ["katex"],
    });
  } catch {
    // Keep the original text if KaTeX cannot parse it.
  }
}
