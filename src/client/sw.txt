self.addEventListener("push", (event) => {
  let data = {};
  try {
    data = event.data ? event.data.json() : {};
  } catch {
    data = {};
  }
  const title = data.title || "糖包";
  const url = typeof data.url === "string" && data.url.startsWith("/") ? data.url : "/desk";
  event.waitUntil((async () => {
    try {
      const target = new URL(url, self.location.origin);
      const targetThread = target.searchParams.get("thread");
      const list = await clients.matchAll({ type: "window", includeUncontrolled: true });
      if (list.some((client) => {
        if (client.visibilityState !== "visible") return false;
        const current = new URL(client.url);
        if (current.pathname !== target.pathname) return false;
        return !targetThread || current.searchParams.get("thread") === targetThread;
      })) return;
    } catch {}
    const options = {
      body: data.body || "",
      icon: "/favicon.svg",
      data: { url },
    };
    if (typeof data.tag === "string" && data.tag) options.tag = data.tag;
    await self.registration.showNotification(title, options);
  })());
});

self.addEventListener("notificationclick", (event) => {
  event.notification.close();
  const url = (event.notification.data && event.notification.data.url) || "/desk";
  event.waitUntil(
    clients.matchAll({ type: "window", includeUncontrolled: true }).then(async (list) => {
      const target = new URL(url, self.location.origin);
      for (const client of list) {
        if (new URL(client.url).pathname !== target.pathname) continue;
        if ("focus" in client) await client.focus();
        if (client.url !== target.href && "navigate" in client) await client.navigate(target.href);
        return;
      }
      return clients.openWindow(url);
    }),
  );
});
