class ApiError extends Error {
  constructor(message, statusCode) {
    super(message);
    this.statusCode = statusCode;
  }
}

async function api(path, options = {}) {
  const form = options.body instanceof FormData;
  let response;
  try {
    response = await fetch(path, {
      method: options.method || "GET",
      headers: options.body && !form ? { "content-type": "application/json" } : undefined,
      body: form ? options.body : options.body ? JSON.stringify(options.body) : undefined,
      credentials: "same-origin",
    });
  } catch {
    throw new ApiError("桌子这边晃了一下。", 0);
  }
  const data = await response.json().catch(() => ({}));
  if (!response.ok) throw new ApiError(data.error || "桌子这边晃了一下。", response.status);
  return data;
}

function spent(ms) {
  const seconds = Math.max(1, Math.round(ms / 1000));
  if (seconds < 60) return `${seconds} 秒`;
  const minutes = Math.floor(seconds / 60);
  if (minutes < 60) return `${minutes} 分钟`;
  const hours = Math.floor(minutes / 60);
  if (hours < 48) return `${hours} 小时${minutes % 60 ? ` ${minutes % 60} 分` : ""}`;
  return `${Math.floor(hours / 24)} 天${hours % 24 ? ` ${hours % 24} 小时` : ""}`;
}

function clock(ts) {
  const date = new Date(ts);
  return `${String(date.getHours()).padStart(2, "0")}:${String(date.getMinutes()).padStart(2, "0")}`;
}

function showDesk(on) {
  gate.hidden = on;
  desk.hidden = !on;
}

function applyAwake(next) {
  awake = next;
  body.dataset.presence = next ? "awake" : "asleep";
  if (seat.super) {
    try {
      localStorage.setItem("tangbao.presence", body.dataset.presence);
    } catch {}
  }
  if (!body.classList.contains("settled")) requestAnimationFrame(() => body.classList.add("settled"));
  lamp.setAttribute("aria-checked", String(next));
  lamp.querySelector("b").textContent = next ? "灯亮着" : "灯灭了";
  lamp.querySelector("small").textContent = next ? "前台显示：可用 · 他醒着" : "前台显示：繁忙 · 他睡了";
}

function slip(item, kind) {
  const button = document.createElement("button");
  button.type = "button";
  button.className = kind === "wait" ? "slip waiting" : "slip";
  if (item.id === selected) button.setAttribute("aria-current", "true");
  const title = document.createElement("b");
  title.textContent = item.title;
  button.append(title);
  const head = item.title.replace(/…$/, "");
  const latest = (item.preview || "").replace(/…$/, "");
  if (latest && !latest.startsWith(head)) {
    const preview = document.createElement("span");
    preview.textContent = item.preview;
    button.append(preview);
  }
  const meta = document.createElement("small");
  meta.textContent = kind === "wait"
    ? `等了 ${spent(Date.now() - item.updatedAt)}`
    : kind === "shelved"
      ? `放下于 ${clock(item.updatedAt)}`
      : kind === "blocked"
        ? `拉黑于 ${clock(item.updatedAt)}`
        : `回于 ${clock(item.updatedAt)}`;
  button.append(meta);
  button.addEventListener("click", () => {
    void openSlip(item.id);
  });
  return button;
}

function renderQueue(data) {
  applyAwake(Boolean(data.awake));
  const waiting = data.waiting || [];
  const recent = data.recent || [];
  count.textContent = waiting.length
    ? `${waiting.length} 条待写。这些 token 还没生成，GPU 是你的手指。`
    : "桌上空着。GPU 在休息。";
  pile.replaceChildren();
  const waitingTitle = document.createElement("h3");
  waitingTitle.textContent = "待你亲自处理";
  pile.append(waitingTitle);
  if (!waiting.length) {
    const none = document.createElement("p");
    none.className = "none";
    none.textContent = "没有新消息。";
    pile.append(none);
  }
  for (const item of waiting) pile.append(slip(item, "wait"));
  if (recent.length) {
    const recentTitle = document.createElement("h3");
    recentTitle.textContent = "刚回过";
    pile.append(recentTitle);
    for (const item of recent) pile.append(slip(item, "done"));
  }
  foldPile("先放下", data.shelved || [], "shelved");
  foldPile("已拉黑", data.blocked || [], "blocked");
}

function foldPile(title, items, kind) {
  if (!items.length) return;
  const details = document.createElement("details");
  details.className = "fold";
  const summary = document.createElement("summary");
  summary.textContent = title;
  details.append(summary);
  for (const item of items) details.append(slip(item, kind));
  pile.append(details);
}

function renderThread(data) {
  $("#slip-title").textContent = data.thread.title;
  const last = data.messages.at(-1);
  const own = data.thread.operatorId === seat.operatorId;
  const locked = readonly || !own;
  const shelve = $("#shelve");
  const block = $("#block");
  shelve.hidden = locked || Boolean(data.thread.blocked) || !data.thread.awaiting;
  block.hidden = !seat.super;
  $("#back-audit").hidden = !readonly;
  reply.hidden = locked;
  if (!data.thread.awaiting) {
    shelve.dataset.confirm = "";
    shelve.textContent = "先放下";
  }
  if (data.thread.blocked) {
    if (block.dataset.confirm !== "1") block.textContent = "解开";
    block.dataset.mode = "undo";
  } else if (block.dataset.confirm !== "1") {
    block.textContent = "拉黑";
    block.dataset.mode = "";
  }
  $("#slip-meta").textContent = data.thread.blocked
    ? "已拉黑"
    : data.thread.awaiting
      ? `等了 ${spent(Date.now() - data.thread.updatedAt)}`
      : last && last.role === "visitor"
        ? "先放下了"
        : "已回复";
  $("#answer").placeholder = last && last.role === "visitor" && last.draw
    ? "他要一张图。画完放进来，或写一句。"
    : "写一句可能没用的话";
  const key = JSON.stringify(data.messages);
  if (key === shown) return;
  shown = key;
  thread.replaceChildren();
  for (const message of data.messages) {
    const wrap = document.createElement("div");
    wrap.className = message.role === "tangbao" ? "me" : "them";
    if (message.body) {
      if (message.role === "tangbao" && typeof renderMarkdown === "function") {
        const text = document.createElement("div");
        text.className = "md";
        text.innerHTML = renderMarkdown(message.body);
        wrap.append(text);
      } else {
        const text = document.createElement("p");
        text.textContent = message.body;
        wrap.append(text);
      }
    }
    const sources = (message.images || []).map((image) => `/api/images/${image.id}`);
    if (sources.length) {
      const shots = document.createElement("div");
      shots.className = "shots";
      for (const src of sources) {
        const img = document.createElement("img");
        img.className = "shot";
        img.alt = "图片，点击放大";
        img.src = src;
        img.tabIndex = 0;
        img.setAttribute("role", "button");
        img.addEventListener("click", () => openViewer(src));
        shots.append(img);
      }
      wrap.append(shots);
    }
    const meta = document.createElement("small");
    const who = message.role === "tangbao" ? (locked ? "糖包" : "你") : "对方";
    meta.textContent = message.draw ? `${who} · 要一张图 · ${clock(message.createdAt)}` : `${who} · ${clock(message.createdAt)}`;
    wrap.append(meta);
    thread.append(wrap);
  }
  thread.scrollTop = thread.scrollHeight;
}

async function loadQueue() {
  const data = await api("/api/admin/queue");
  seat = { super: Boolean(data.super), operatorId: data.operatorId || "" };
  showDesk(true);
  $("#desk-nav").hidden = !seat.super;
  renderQueue(data);
}

function showPane(name, nav) {
  $("#pane-thread").hidden = name !== "thread";
  $("#pane-apply").hidden = name !== "apply";
  $("#pane-audit").hidden = name !== "audit";
  const current = nav || name;
  for (const button of $("#desk-nav").querySelectorAll("button")) {
    button.setAttribute("aria-pressed", String(button.dataset.view === current));
  }
}

async function openSlip(id, fromAudit) {
  readonly = Boolean(fromAudit);
  showPane("thread", readonly ? "audit" : "thread");
  if (id !== selected) shown = "";
  selected = id;
  replyErr.textContent = "";
  try {
    renderThread(await api(`/api/admin/threads/${id}`));
    await loadQueue();
  } catch (err) {
    replyErr.textContent = err.message;
  }
  if (window.matchMedia("(max-width: 860px)").matches) $("#reader").scrollIntoView({ behavior: "smooth" });
  if (!reply.hidden) answer.focus();
}

function arm() {
  clearTimeout(timer);
  if (document.hidden || desk.hidden) return;
  timer = setTimeout(() => {
    void tick();
  }, POLL_MS);
}

async function tick() {
  try {
    await loadQueue();
    if (selected && !$("#pane-thread").hidden) renderThread(await api(`/api/admin/threads/${selected}`));
    if (!$("#pane-apply").hidden) await loadApplications();
    if (!$("#pane-audit").hidden) await loadAudit();
  } catch (err) {
    if (err instanceof ApiError && err.statusCode === 401) {
      showDesk(false);
      return;
    }
  }
  arm();
}

function statusWord(status) {
  if (status === "approved") return "过了";
  if (status === "rejected") return "没过";
  return "在等";
}

function confirmClick(button, idle, armed) {
  if (button.dataset.confirm !== "1") {
    button.dataset.confirm = "1";
    button.textContent = armed;
    window.setTimeout(() => {
      if (button.dataset.confirm !== "1") return;
      button.dataset.confirm = "";
      button.textContent = idle;
    }, 2200);
    return false;
  }
  button.dataset.confirm = "";
  button.textContent = idle;
  return true;
}

async function loadApplications() {
  const data = await api("/api/admin/applications");
  const box = $("#applications");
  box.replaceChildren();
  const rows = data.applications || [];
  if (!rows.length) {
    const empty = document.createElement("p");
    empty.className = "quiet";
    empty.textContent = "没有申请。";
    box.append(empty);
    return;
  }
  for (const item of rows) {
    const paper = document.createElement("article");
    paper.className = "paper";
    const title = document.createElement("b");
    title.textContent = `tangbao-${item.suffix}`;
    paper.append(title);
    const meta = document.createElement("p");
    meta.textContent = `${item.email} · ${statusWord(item.status)}`;
    paper.append(meta);
    if (item.status === "rejected" && item.reason) {
      const why = document.createElement("p");
      why.textContent = item.reason;
      paper.append(why);
    }
    const password = issued.get(item.id);
    if (password) {
      const once = document.createElement("p");
      once.innerHTML = `口令 <span class="pass"></span>。只出现这一次，交给对方。对方到 /desk 用它坐下。`;
      once.querySelector(".pass").textContent = password;
      paper.append(once);
    }
    if (item.status === "pending") {
      const acts = document.createElement("div");
      acts.className = "decide";
      const approve = document.createElement("button");
      approve.type = "button";
      approve.className = "yes";
      approve.textContent = "通过";
      approve.addEventListener("click", async () => {
        if (!confirmClick(approve, "通过", "确定通过")) return;
        try {
          const result = await api(`/api/admin/applications/${item.id}/approve`, { method: "POST" });
          issued.set(item.id, result.password);
          await loadApplications();
        } catch (err) {
          meta.textContent = err.message;
        }
      });
      const reject = document.createElement("button");
      reject.type = "button";
      reject.className = "no";
      reject.textContent = "没过";
      reject.addEventListener("click", () => {
        if (paper.querySelector(".reason")) return;
        const box = document.createElement("div");
        box.className = "reason";
        const note = document.createElement("textarea");
        note.rows = 2;
        note.maxLength = 200;
        note.placeholder = "写一句没过的理由，申请人能看见";
        const row = document.createElement("div");
        row.className = "decide";
        const send = document.createElement("button");
        send.type = "button";
        send.className = "no";
        send.textContent = "确定没过";
        const cancel = document.createElement("button");
        cancel.type = "button";
        cancel.className = "quiet";
        cancel.textContent = "取消";
        cancel.addEventListener("click", () => box.remove());
        send.addEventListener("click", async () => {
          const reason = note.value.trim();
          if (!reason) {
            meta.textContent = "先写一句没过的理由。";
            return;
          }
          try {
            await api(`/api/admin/applications/${item.id}/reject`, { method: "POST", body: { reason } });
            await loadApplications();
          } catch (err) {
            meta.textContent = err.message;
          }
        });
        row.append(send, cancel);
        box.append(note, row);
        acts.after(box);
        note.focus();
      });
      acts.append(approve, reject);
      paper.append(acts);
    }
    box.append(paper);
  }
}

async function loadAudit() {
  const data = await api("/api/admin/audit");
  const box = $("#audit");
  box.replaceChildren();
  const models = document.createElement("h3");
  models.textContent = "型号";
  box.append(models);
  const operators = data.operators || [];
  if (!operators.length) {
    const empty = document.createElement("p");
    empty.className = "quiet";
    empty.textContent = "还没有别的型号。";
    box.append(empty);
  }
  for (const operator of operators) {
    const paper = document.createElement("article");
    paper.className = "paper";
    const title = document.createElement("b");
    title.textContent = operator.id;
    paper.append(title);
    const meta = document.createElement("p");
    meta.textContent = operator.active ? (operator.awake ? "在用 · 灯亮着" : "在用 · 灯灭了") : "停了";
    paper.append(meta);
    const toggle = document.createElement("button");
    toggle.type = "button";
    toggle.className = "gate-switch";
    toggle.setAttribute("role", "switch");
    toggle.setAttribute("aria-checked", operator.active ? "true" : "false");
    toggle.innerHTML = '<span class="track" aria-hidden="true"></span><b></b>';
    const label = toggle.querySelector("b");
    label.textContent = operator.active ? "在用" : "停了";
    toggle.addEventListener("click", async () => {
      const on = toggle.getAttribute("aria-checked") === "true";
      if (on && toggle.dataset.confirm !== "1") {
        toggle.dataset.confirm = "1";
        label.textContent = "确定停用";
        window.setTimeout(() => {
          if (toggle.dataset.confirm !== "1") return;
          toggle.dataset.confirm = "";
          label.textContent = "在用";
        }, 2200);
        return;
      }
      toggle.dataset.confirm = "";
      try {
        await api(`/api/admin/operators/${operator.id}/${on ? "disable" : "enable"}`, { method: "POST" });
        await loadAudit();
      } catch (err) {
        meta.textContent = err.message;
      }
    });
    paper.append(toggle);
    box.append(paper);
  }
  const talks = document.createElement("h3");
  talks.textContent = "对话";
  box.append(talks);
  const threads = data.threads || [];
  if (!threads.length) {
    const empty = document.createElement("p");
    empty.className = "quiet";
    empty.textContent = "还没有对话。";
    box.append(empty);
  }
  for (const item of threads) {
    const button = document.createElement("button");
    button.type = "button";
    button.className = "audit-line";
    const title = document.createElement("b");
    title.textContent = item.title || "没有标题";
    const who = document.createElement("span");
    who.textContent = item.operatorId;
    const meta = document.createElement("small");
    meta.textContent = item.awaiting ? "在等" : clock(item.updatedAt);
    button.append(title, who, meta);
    button.addEventListener("click", () => {
      void openSlip(item.id, true);
    });
    box.append(button);
  }
}

reply.addEventListener("submit", async (event) => {
  event.preventDefault();
  if (!selected || replying) return;
  const text = answer.value.trim();
  const images = pendingImages.slice();
  if (!text && !images.length) {
    replyErr.textContent = "写一句可能没用的话，或放图片。";
    return;
  }
  const key = `${text}\n${images.map((item) => `${item.file.size}:${item.file.lastModified}`).join(",")}`;
  const now = Date.now();
  if (replyGuard.key === key && now - replyGuard.at < 1500) return;
  replying = true;
  replyGuard = { key, at: now };
  const button = reply.querySelector(".send");
  button.disabled = true;
  replyErr.textContent = "";
  const payload = images.length ? new FormData() : { body: text };
  if (images.length) {
    payload.set("body", text);
    for (const item of images) payload.append("images", item.file, item.file.name);
  }
  try {
    await api(`/api/admin/threads/${selected}`, { method: "POST", body: payload });
    answer.value = "";
    clearAttach();
    shown = "";
    renderThread(await api(`/api/admin/threads/${selected}`));
    await loadQueue();
  } catch (err) {
    replyGuard.key = "";
    replyErr.textContent = err.message;
  } finally {
    replying = false;
    button.disabled = false;
  }
});

async function shrinkImage(file) {
  if (!["image/jpeg", "image/png", "image/webp", "image/gif"].includes(file.type)) {
    throw new Error("这张图打不开。换 jpg、png、webp 或 gif。");
  }
  if (file.type === "image/gif") {
    if (file.size > 2_000_000) throw new Error("这张图太大了。先收一收再发。");
    return file;
  }
  const bitmap = await createImageBitmap(file);
  const scale = Math.min(1, 1600 / Math.max(bitmap.width, bitmap.height));
  const canvas = document.createElement("canvas");
  canvas.width = Math.max(1, Math.round(bitmap.width * scale));
  canvas.height = Math.max(1, Math.round(bitmap.height * scale));
  canvas.getContext("2d").drawImage(bitmap, 0, 0, canvas.width, canvas.height);
  bitmap.close();
  const blob = await new Promise((resolve) => canvas.toBlob(resolve, "image/jpeg", 0.82));
  if (!blob || blob.size > 2_000_000) throw new Error("这张图太大了。先收一收再发。");
  return new File([blob], "image.jpg", { type: "image/jpeg" });
}

function showAttach() {
  const box = $("#attach");
  box.replaceChildren();
  if (!pendingImages.length) {
    box.hidden = true;
    return;
  }
  pendingImages.forEach((item, index) => {
    const thumb = document.createElement("div");
    thumb.className = "thumb";
    const img = document.createElement("img");
    img.src = item.url;
    img.alt = "";
    const remove = document.createElement("button");
    remove.type = "button";
    remove.textContent = "×";
    remove.setAttribute("aria-label", "拿掉这张图");
    remove.addEventListener("click", () => {
      const [gone] = pendingImages.splice(index, 1);
      if (gone) URL.revokeObjectURL(gone.url);
      showAttach();
    });
    thumb.append(img, remove);
    box.append(thumb);
  });
  box.hidden = false;
}

function clearAttach() {
  for (const item of pendingImages) URL.revokeObjectURL(item.url);
  pendingImages = [];
  const file = $("#file");
  if (file) file.value = "";
  showAttach();
}

async function addImages(files) {
  const list = [...files].filter(Boolean);
  if (!list.length) return;
  if (pendingImages.length + list.length > MAX_IMAGES) replyErr.textContent = `一次最多 ${MAX_IMAGES} 张。`;
  const room = MAX_IMAGES - pendingImages.length;
  for (const file of list.slice(0, room)) {
    const shrunk = await shrinkImage(file);
    pendingImages.push({ file: shrunk, url: URL.createObjectURL(shrunk) });
  }
  showAttach();
}

const sketch = $("#sketch");
const pad = $("#sketch-pad");
const pen = pad.getContext("2d");
const PAPER = "#fffaf3";
let strokes = [];
let currentStroke = null;
let penSize = 6;
let penColor = "#1f1d1a";
let erasing = false;
let padSize = { w: 1, h: 1, dpr: 1 };

function preparePad() {
  const width = Math.min(720, window.innerWidth - 64);
  const height = Math.min(420, Math.round(window.innerHeight * 0.52));
  const dpr = Math.min(window.devicePixelRatio || 1, 2);
  padSize = { w: width, h: height, dpr };
  pad.width = Math.round(width * dpr);
  pad.height = Math.round(height * dpr);
  pen.setTransform(dpr, 0, 0, dpr, 0, 0);
  pen.lineCap = "round";
  pen.lineJoin = "round";
  redrawPad();
}

function redrawPad() {
  pen.setTransform(padSize.dpr, 0, 0, padSize.dpr, 0, 0);
  pen.fillStyle = PAPER;
  pen.fillRect(0, 0, padSize.w, padSize.h);
  pen.lineCap = "round";
  pen.lineJoin = "round";
  for (const stroke of strokes) paintStroke(stroke);
  if (currentStroke) paintStroke(currentStroke);
}

function paintStroke(stroke) {
  if (!stroke.points.length) return;
  pen.strokeStyle = stroke.color;
  pen.fillStyle = stroke.color;
  pen.lineWidth = stroke.size;
  const first = stroke.points[0];
  if (stroke.points.length === 1) {
    pen.beginPath();
    pen.arc(first.x, first.y, stroke.size / 2, 0, Math.PI * 2);
    pen.fill();
    return;
  }
  pen.beginPath();
  pen.moveTo(first.x, first.y);
  for (let i = 1; i < stroke.points.length; i += 1) pen.lineTo(stroke.points[i].x, stroke.points[i].y);
  pen.stroke();
}

function padPoint(event) {
  const rect = pad.getBoundingClientRect();
  return {
    x: (event.clientX - rect.left) * (padSize.w / rect.width),
    y: (event.clientY - rect.top) * (padSize.h / rect.height),
  };
}

function openSketch() {
  closePlus();
  strokes = [];
  currentStroke = null;
  erasing = false;
  $("#sketch-erase").setAttribute("aria-pressed", "false");
  $("#sketch-err").textContent = "";
  sketch.hidden = false;
  requestAnimationFrame(preparePad);
}

function closeSketch() {
  sketch.hidden = true;
}

pad.addEventListener("pointerdown", (event) => {
  if (sketch.hidden) return;
  $("#sketch-err").textContent = "";
  pad.setPointerCapture(event.pointerId);
  currentStroke = {
    color: erasing ? PAPER : penColor,
    size: erasing ? Math.max(penSize * 4, 16) : penSize,
    points: [padPoint(event)],
  };
  redrawPad();
});
pad.addEventListener("pointermove", (event) => {
  if (!currentStroke || !pad.hasPointerCapture(event.pointerId)) return;
  currentStroke.points.push(padPoint(event));
  redrawPad();
});
function endStroke(event) {
  if (!currentStroke) return;
  if (event && pad.hasPointerCapture(event.pointerId)) pad.releasePointerCapture(event.pointerId);
  if (currentStroke.points.length) strokes.push(currentStroke);
  currentStroke = null;
}
pad.addEventListener("pointerup", endStroke);
pad.addEventListener("pointercancel", endStroke);
