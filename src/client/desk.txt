class ApiError extends Error {
  constructor(message, statusCode, code) {
    super(message);
    this.statusCode = statusCode;
    this.code = code;
  }
}

async function api(path, options = {}) {
  if (loggedOut) throw new ApiError("此设备已退出，请重新登录。", 401, "LOCAL_LOGOUT");
  const accountId = TangbaoAccount.account?.id || null;
  const form = options.body instanceof FormData;
  let response;
  try {
    response = await fetch(path, {
      method: options.method || "GET",
      headers: options.body && !form ? { "content-type": "application/json" } : undefined,
      body: form ? options.body : options.body ? JSON.stringify(options.body) : undefined,
      credentials: "same-origin",
    });
  } catch {
    throw new ApiError("笼屉晃了一下。", 0);
  }
  const data = await response.json().catch(() => ({}));
  if (loggedOut) throw new ApiError("此设备已退出，请重新登录。", 401, "LOCAL_LOGOUT");
  if (accountId !== (TangbaoAccount.account?.id || null)) throw new ApiError("账号已切换，请重新打开。", 401, "AUTH_CONTEXT_CHANGED");
  if (!response.ok) throw new ApiError(data.error || "笼屉晃了一下。", response.status, data.code);
  return data;
}

function spent(ms) {
  const seconds = Math.max(1, Math.round(ms / 1000));
  if (seconds < 60) return `${seconds} 秒`;
  const minutes = Math.floor(seconds / 60);
  if (minutes < 60) return `${minutes} 分钟`;
  const hours = Math.floor(minutes / 60);
  if (hours < 48) return `${hours} 小时${minutes % 60 ? ` ${minutes % 60} 分` : ""}`;
  return `${Math.floor(hours / 24)} 天${hours % 24 ? ` ${hours % 24} 小时` : ""}`;
}

function clock(ts) {
  const date = new Date(ts);
  return `${String(date.getHours()).padStart(2, "0")}:${String(date.getMinutes()).padStart(2, "0")}`;
}

function when(ts) {
  const date = new Date(ts);
  const time = clock(ts);
  if (date.toDateString() === new Date().toDateString()) return time;
  return `${date.getMonth() + 1}/${date.getDate()} ${time}`;
}

}

async function loadQueue() {
  if (!heartbeatTimer && TangbaoAccount.account?.verified && !document.hidden) await sendPresence(false);
  const data = await api("/api/admin/queue");
  waiting = (data.waiting || []).length;
  unread = (data.waiting || []).filter((item) => item.unread).length;
  pending = Number(data.pending) || 0;
  seat = { super: Boolean(data.super), operatorId: data.operatorId || "" };
  workState.hidden = !seat.operatorId;
  applyWorkState(data.accepting !== false);
  startPresenceHeartbeat();
  if (!document.hidden && data.status?.status === "away" && !presenceWarmup) {
    presenceWarmup = true;
    window.setTimeout(() => {
      if (!document.hidden) void loadQueue();
    }, 300);
  }
  showDesk(true);
  $("#desk-nav").hidden = !seat.super;
  renderQueue(data);
  paintTitle();
  if (profileRequested) {
    profileRequested = false;
    if (location.search) window.history.replaceState(null, "", "/desk");
    if (seat.operatorId && seat.operatorId !== "tangbao-vme50-flash") void openProfilePane();
  } else if (!deepLinkOpened && deepLinkThread) {
    deepLinkOpened = true;
    if (/^[A-Za-z0-9_-]{16,80}$/.test(deepLinkThread)) void openSlip(deepLinkThread);
  }
}

async function saveProfile(event) {
  event.preventDefault();
  profileErr.textContent = "";
  profileOk.textContent = "";
  const button = profileForm.querySelector("button[type=submit]");
  button.disabled = true;
  try {
    const oldId = seat.operatorId;
    const data = await api("/api/admin/profile", {
      method: "POST",
      body: {
        suffix: profileSuffix.value.trim(),
        blurb: profileBlurb.value.trim(),
        openingQuestions: openingFields().map((field) => field.value.trim()).filter(Boolean),
      },
    });
    seat.operatorId = data.id;
    profileSuffix.value = data.suffix;
    profileBlurb.value = data.blurb;
    paintOpeningRows(Array.isArray(data.openingQuestions) ? data.openingQuestions : []);
    profileOk.textContent = data.id === oldId ? "资料换好了。" : "型号换好了。旧对话都还在。";
    if (data.id !== oldId) {
      selected = null;
      shown = "";
      $("#slip-title").textContent = "还没选消息";
      $("#slip-meta").textContent = "";
      thread.replaceChildren();
      reply.hidden = true;
      await loadQueue();
    }
  } catch (err) {
    profileErr.textContent = err.message;
  } finally {
    button.disabled = false;
  }
}

function setDeskThreadUrl(id) {
  const next = id ? `/desk?thread=${encodeURIComponent(id)}` : "/desk";
  if (`${location.pathname}${location.search}` !== next) {
    try { window.history.replaceState(null, "", next); } catch {}

}

async function closeDeskThreadNotification(id) {
  if (!("serviceWorker" in navigator)) return;
  try {
    const registration = await navigator.serviceWorker.getRegistration();
    const notifications = await registration?.getNotifications({ tag: `desk:${id}` }) || [];
    for (const notification of notifications) notification.close();
  } catch {}
}

async function openSlip(id, fromAudit) {
  TangbaoReports.cancelReview(thread);
  activeReportId = null;
  const version = ++threadViewVersion;
  readonly = Boolean(fromAudit);
  setDeskThreadUrl(id);
  void closeDeskThreadNotification(id);
  showPane("thread", readonly ? "audit" : "thread");
  if (id !== selected) {
    shown = "";
    reply.hidden = true;
    $("#block").hidden = true;
    $("#shelve").hidden = true;
    thread.replaceChildren();
    $("#slip-title").textContent = "正在读取…";
  }
  selected = id;
  replyErr.textContent = "";
  try {
    const data = await messagePages.load(`${version}:${id}`, `/api/admin/threads/${id}`);
    if (selected !== id || version !== threadViewVersion) return;
    renderThread(data);
    if (!readonly && data.thread.operatorId === seat.operatorId) {
      await api(`/api/admin/threads/${id}/read`, { method: "POST" }).catch(() => {});
    }
    await loadQueue();
  } catch (err) {
    if (version !== threadViewVersion) return;
    replyErr.textContent = err.message;
    $("#slip-title").textContent = "打不开这条对话";
    $("#slip-meta").textContent = err.message;
    reply.hidden = true;
    thread.replaceChildren();
    const note = document.createElement("p");
    note.className = "quiet";
    note.textContent = "这条对话暂时打不开。可以点左上角刷新后再试。";
    thread.append(note);
  }
  if (version !== threadViewVersion) return;
  if (window.matchMedia("(max-width: 860px)").matches) $("#reader").scrollIntoView({ behavior: "smooth" });
  if (!reply.hidden) answer.focus();
}

function sendPresence(away, preferBeacon = false, graceMs = 0) {
  if (!TangbaoAccount.account?.verified) return Promise.resolve();
  const body = JSON.stringify({ away, graceMs });
  if (preferBeacon && navigator.sendBeacon) {
    try {
      if (navigator.sendBeacon("/api/admin/presence", new Blob([body], { type: "application/json" }))) return Promise.resolve();
    } catch {}
  }
  return fetch("/api/admin/presence", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body,
    credentials: "same-origin",
    keepalive: away,
  }).catch(() => {});
}

  heartbeatTimer = 0;

}

function arm() {
  clearTimeout(timer);
  if (desk.hidden || document.hidden) return;
  const fresh = waiting > 0 || pending > 0;
  const wait = fresh ? 30000 : POLL_MS;
  timer = setTimeout(() => {
    void tick();
  }, wait);
}

async function tick() {
  if (document.hidden) return;
  const account = TangbaoAccount.account;
  if (!account?.verified || (!account.operatorId && !account.roles?.includes("admin"))) return;
  if (ticking && pollingAccountId === account.id) return ticking;
  clearTimeout(timer);
  const request = (async () => {
    try {
      await loadQueue();
      const id = selected;
      const version = threadViewVersion;
      if (id && !activeReportId && !$("#pane-thread").hidden) {
        const data = await messagePages.load(`${version}:${id}`, `/api/admin/threads/${id}`);
        if (selected === id && version === threadViewVersion && !$("#pane-thread").hidden) renderThread(data);
      }
      if (!$("#pane-apply").hidden) await loadApplications();
      if (!$("#pane-audit").hidden) await loadAudit();
    } catch (err) {
      if (account.id !== TangbaoAccount.account?.id) return;
      if (err instanceof ApiError && (err.statusCode === 401 || err.statusCode === 403)) {
        resetDeskView();
        loginErr.textContent = err.message;
        $("#gate-note").textContent = "当前账号暂时无法进入笼屉，请根据提示处理。";
        await TangbaoAccount.refresh().catch(() => null);
      }
    } finally {
      if (ticking === request) {
        ticking = null;
        arm();
      }
    }
  })();
  ticking = request;
  pollingAccountId = account.id;
  return request;
}

function confirmClick(button, idle, armed) {
  if (button.dataset.confirm !== "1") {
    button.dataset.confirm = "1";
    button.textContent = armed;
    window.setTimeout(() => {
      if (button.dataset.confirm !== "1") return;
      button.dataset.confirm = "";

}

function paintPager(parent, page, total, size, move, unit = "") {
  const pages = Math.max(1, Math.ceil((Number(total) || 0) / (Number(size) || 10)));
  if (pages <= 1) return;
  const nav = document.createElement("nav");
  nav.className = "pager";
  nav.setAttribute("aria-label", "分页");
  const info = document.createElement("span");
  info.className = "pager-info";
  const current = document.createElement("b");
  current.textContent = String(page);
  info.append("第 ", current, ` / ${pages} 页 · 共 ${total} ${unit}`);
  const actions = document.createElement("div");
  actions.className = "pager-actions";
  const prev = document.createElement("button");
  prev.type = "button";
  prev.textContent = "上一页";
  prev.disabled = page <= 1;
  prev.addEventListener("click", () => move(page - 1));
  const next = document.createElement("button");
  next.type = "button";
  next.textContent = "下一页";
  next.disabled = page >= pages;
  next.addEventListener("click", () => move(page + 1));
  actions.append(prev, next);
  nav.append(info, actions);
  parent.append(nav);
}

function applicationCard(item) {
  const paper = document.createElement("article");
  paper.className = "paper";
  paper.dataset.status = item.status;

  const body = document.createElement("div");
  body.className = "body";
  const head = document.createElement("div");
  head.className = "head";
  const title = document.createElement("b");
  title.textContent = `tangbao-${item.suffix}`;
  title.title = title.textContent;
  head.append(title);
  body.append(head);

  if (item.blurb) {
    const blurb = document.createElement("p");
    blurb.className = "blurb";
    blurb.textContent = item.blurb;
    body.append(blurb);
  } else {
    const blurb = document.createElement("p");
    blurb.className = "blurb empty";
    blurb.textContent = "这一份没写说明。";
    body.append(blurb);
  }

  const meta = document.createElement("p");
  meta.className = "mail";
  meta.textContent = item.email;
  if (item.createdAt) {
    if (item.status === "pending") {
      const wait = document.createElement("span");
      wait.className = "wait";
      wait.textContent = ` · 等了 ${spent(Date.now() - item.createdAt)}`;
      meta.append(wait, ` · ${when(item.createdAt)}`);
    } else {
      meta.append(` · ${when(item.createdAt)}`);
    }
  }
  body.append(meta);

  if (item.status === "rejected") {
    const reason = String(item.reason || "").trim();
    const why = document.createElement("p");
    why.className = reason ? "why" : "why empty";
    const label = document.createElement("b");
    label.textContent = "没过的理由";
    why.append(label);
    if (reason) {
      why.append(document.createTextNode(reason));
    } else {
      const none = document.createElement("span");
      none.className = "why-none";
      none.textContent = "当时没写理由。";
      why.append(none);
    }
    body.append(why);
  }

  const side = document.createElement("div");
  side.className = "paper-side";
  const state = document.createElement("span");
  state.className = "state";
  state.textContent = statusWord(item.status);
  side.append(state);

  if (item.status === "pending") {
    const acts = document.createElement("div");
    acts.className = "decide";
    const approve = document.createElement("button");
    approve.type = "button";
    approve.className = "yes";
    approve.textContent = "通过";
    approve.addEventListener("click", async () => {
      if (!confirmClick(approve, "通过", "确定通过")) return;
      try {
        await api(`/api/admin/applications/${item.id}/approve`, { method: "POST" });
        await loadApplications(true);
      } catch (err) {
        paperNote(paper, err.message);
      }
    });
    const reject = document.createElement("button");
    reject.type = "button";
    reject.className = "no";
    reject.textContent = "没过";
    reject.addEventListener("click", () => {
      if (paper.querySelector(".reject-box")) return;
      const form = document.createElement("div");
      form.className = "reject-box";
      const note = document.createElement("textarea");
      note.rows = 2;
      note.maxLength = 200;
      note.placeholder = "写一句没过的理由，申请人能看见";
      const row = document.createElement("div");
      row.className = "decide";
      const send = document.createElement("button");
      send.type = "button";
      send.className = "yes";
      send.textContent = "确定没过";
      const cancel = document.createElement("button");
      cancel.type = "button";
      cancel.className = "quiet";
      cancel.textContent = "取消";
      cancel.addEventListener("click", () => form.remove());
      send.addEventListener("click", async () => {
        const reason = note.value.trim();
        if (!reason) {
          paperNote(paper, "先写一句没过的理由。");
          return;
        }
        try {
          await api(`/api/admin/applications/${item.id}/reject`, { method: "POST", body: { reason } });
          await loadApplications(true);
        } catch (err) {
          paperNote(paper, err.message);
        }
      });
      row.append(send, cancel);
      form.append(note, row);
      paper.append(form);
      note.focus();
    });
    acts.append(approve, reject);
    side.append(acts);
  }

  paper.append(body, side);

  if (item.status === "approved") {
    const note = document.createElement("p"); note.className = "once-note";

  }

  return paper;
}

async function shrinkImage(file) {
  if (!["image/jpeg", "image/png", "image/webp", "image/gif"].includes(file.type)) {
    throw new Error("这张图打不开。换 jpg、png、webp 或 gif。");
  }
  if (file.type === "image/gif") {
    if (file.size > 2_000_000) throw new Error("这张图太大了。先收一收再发。");
    return file;
  }
  const bitmap = await createImageBitmap(file);
  const scale = Math.min(1, 1600 / Math.max(bitmap.width, bitmap.height));
  const canvas = document.createElement("canvas");
  canvas.width = Math.max(1, Math.round(bitmap.width * scale));
  canvas.height = Math.max(1, Math.round(bitmap.height * scale));
  canvas.getContext("2d").drawImage(bitmap, 0, 0, canvas.width, canvas.height);
  bitmap.close();
  const blob = await new Promise((resolve) => canvas.toBlob(resolve, "image/jpeg", 0.82));
  if (!blob || blob.size > 2_000_000) throw new Error("这张图太大了。先收一收再发。");
  return new File([blob], "image.jpg", { type: "image/jpeg" });
}

const sketch = $("#sketch");

const PAPER = "#fffaf3";
const FILL_TOLERANCE = 48;
const ZOOM_STEPS = [0.5, 0.75, 1, 1.25, 1.5, 2, 2.5];
let actions = [];

let penSize = 6;
let penColor = "#1f1d1a";

function preparePad() {
  const width = Math.min(720, window.innerWidth - 64);
  const height = Math.min(420, Math.round(window.innerHeight * 0.52));
  const dpr = Math.min(window.devicePixelRatio || 1, 2);
  padSize = { w: width, h: height, dpr };
  pad.width = Math.round(width * dpr);
  pad.height = Math.round(height * dpr);
  baseCanvas.width = pad.width;
  baseCanvas.height = pad.height;
  applyZoomSize();
  pen.setTransform(dpr, 0, 0, dpr, 0, 0);
  pen.lineCap = "round";
  pen.lineJoin = "round";
  basePen.setTransform(dpr, 0, 0, dpr, 0, 0);
  basePen.lineCap = "round";
  basePen.lineJoin = "round";
  renderBase();
  paintSketch();
}

function applyZoomSize() {
  pad.style.width = `${padSize.w * zoom}px`;
  pad.style.height = `${padSize.h * zoom}px`;
  updateZoomControls();
}

function updateZoomControls() {
  const percent = `${Math.round(zoom * 100)}%`;
  $("#sketch-zoom-reset").textContent = percent;
  $("#sketch-zoom-reset").setAttribute("aria-label", `当前缩放 ${percent}，点击恢复 100%`);
  $("#sketch-zoom-out").disabled = zoom <= ZOOM_STEPS[0];
  $("#sketch-zoom-in").disabled = zoom >= ZOOM_STEPS[ZOOM_STEPS.length - 1];
}

function applyZoom(next, anchorX, anchorY) {
  if (currentStroke) return;
  const target = Math.max(ZOOM_STEPS[0], Math.min(ZOOM_STEPS[ZOOM_STEPS.length - 1], next));
  if (Math.abs(target - zoom) < 0.001) return;
  const oldWidth = padSize.w * zoom;
  const oldHeight = padSize.h * zoom;
  const viewX = anchorX ?? sketchViewport.clientWidth / 2;
  const viewY = anchorY ?? sketchViewport.clientHeight / 2;
  const worldX = ((sketchViewport.scrollLeft + viewX) / oldWidth) * padSize.w;
  const worldY = ((sketchViewport.scrollTop + viewY) / oldHeight) * padSize.h;
  zoom = target;
  applyZoomSize();
  sketchViewport.scrollLeft = (worldX / padSize.w) * padSize.w * zoom - viewX;
  sketchViewport.scrollTop = (worldY / padSize.h) * padSize.h * zoom - viewY;
}

function zoomStep(direction) {
  const index = ZOOM_STEPS.findIndex((value) => Math.abs(value - zoom) < 0.001);
  const next = ZOOM_STEPS[Math.max(0, Math.min(ZOOM_STEPS.length - 1, (index < 0 ? 2 : index) + direction))];
  applyZoom(next);
}

function paintPaper(ctx) {
  ctx.setTransform(1, 0, 0, 1, 0, 0);
  ctx.fillStyle = PAPER;
  ctx.fillRect(0, 0, ctx.canvas.width, ctx.canvas.height);
  ctx.setTransform(padSize.dpr, 0, 0, padSize.dpr, 0, 0);
  ctx.lineCap = "round";
  ctx.lineJoin = "round";
}

function renderBase() {
  paintPaper(basePen);
  for (let i = 0; i < cursor; i += 1) {
    const action = actions[i];
    if (action.type === "stroke") paintStroke(basePen, action.stroke);
    else if (action.type === "fill") applyFill(basePen, action);
    else if (action.type === "clear") paintPaper(basePen);
  }
}

function paintSketch() {
  pen.setTransform(1, 0, 0, 1, 0, 0);
  pen.clearRect(0, 0, pad.width, pad.height);
  pen.drawImage(baseCanvas, 0, 0);
  pen.setTransform(padSize.dpr, 0, 0, padSize.dpr, 0, 0);
  pen.lineCap = "round";
  pen.lineJoin = "round";
  if (currentStroke) paintStroke(pen, currentStroke);
}

function commit(action) {
  actions.length = cursor;
  actions.push(action);
  cursor = actions.length;
}

function paintStroke(ctx, stroke) {
  if (!stroke.points.length) return;
  ctx.strokeStyle = stroke.color;
  ctx.fillStyle = stroke.color;
  ctx.lineWidth = stroke.size;
  const first = stroke.points[0];
  if (stroke.points.length === 1) {
    ctx.beginPath();
    ctx.arc(first.x, first.y, stroke.size / 2, 0, Math.PI * 2);
    ctx.fill();
    return;
  }
  ctx.beginPath();
  ctx.moveTo(first.x, first.y);
  for (let i = 1; i < stroke.points.length - 1; i += 1) {
    const point = stroke.points[i];
    const next = stroke.points[i + 1];
    ctx.quadraticCurveTo(point.x, point.y, (point.x + next.x) / 2, (point.y + next.y) / 2);
  }
  const last = stroke.points[stroke.points.length - 1];
  ctx.lineTo(last.x, last.y);
  ctx.stroke();
}

function rgbOf(hex) {
  const value = String(hex).replace("#", "");
  if (!/^[0-9a-f]{6}$/i.test(value)) return [31, 29, 26];
  return [parseInt(value.slice(0, 2), 16), parseInt(value.slice(2, 4), 16), parseInt(value.slice(4, 6), 16)];
}

function floodFill(ctx, seedX, seedY, color, tolerance) {
  const width = ctx.canvas.width;
  const height = ctx.canvas.height;
  if (seedX < 0 || seedY < 0 || seedX >= width || seedY >= height) return false;
  const image = ctx.getImageData(0, 0, width, height);
  const data = image.data;
  const seed = (seedY * width + seedX) * 4;
  const sr = data[seed];
  const sg = data[seed + 1];
  const sb = data[seed + 2];
  const [fr, fg, fb] = rgbOf(color);
  if (sr === fr && sg === fg && sb === fb) return false;
  const matches = (index) => Math.max(Math.abs(data[index] - sr), Math.abs(data[index + 1] - sg), Math.abs(data[index + 2] - sb)) <= tolerance;
  const seen = new Uint8Array(width * height);
  const stack = [[seedX, seedY]];
  let changed = 0;
  while (stack.length) {
    const [startX, y] = stack.pop();
    let x = startX;
    while (x >= 0 && !seen[y * width + x] && matches((y * width + x) * 4)) x -= 1;
    x += 1;
    let up = false;
    let down = false;
    while (x < width && !seen[y * width + x] && matches((y * width + x) * 4)) {
      const pixel = y * width + x;
      seen[pixel] = 1;
      const index = pixel * 4;
      data[index] = fr;
      data[index + 1] = fg;
      data[index + 2] = fb;
      data[index + 3] = 255;
      changed += 1;
      if (y > 0) {
        const above = (y - 1) * width + x;
        if (!seen[above] && matches(above * 4)) {
          if (!up) { stack.push([x, y - 1]); up = true; }
        } else {
          up = false;
        }
      }
      if (y < height - 1) {
        const below = (y + 1) * width + x;
        if (!seen[below] && matches(below * 4)) {
          if (!down) { stack.push([x, y + 1]); down = true; }
        } else {
          down = false;
        }
      }
      x += 1;
    }
  }
  if (!changed) return false;
  ctx.putImageData(image, 0, 0);
  return true;
}

function applyFill(ctx, action) {
  floodFill(ctx, Math.round(action.x * padSize.dpr), Math.round(action.y * padSize.dpr), action.color, action.tolerance ?? FILL_TOLERANCE);
}

function fillAt(x, y) {
  if (currentStroke) return;
  const color = penColor;
  if (!floodFill(basePen, Math.round(x * padSize.dpr), Math.round(y * padSize.dpr), color, FILL_TOLERANCE)) return;
  commit({ type: "fill", x, y, color, tolerance: FILL_TOLERANCE });
  paintSketch();
}

function hasContent() {
  return cursor > 0 && actions[cursor - 1].type !== "clear";
}

function padPoint(event) {
  const rect = pad.getBoundingClientRect();
  return {
    x: (event.clientX - rect.left) * (padSize.w / rect.width),
    y: (event.clientY - rect.top) * (padSize.h / rect.height),
  };
}

function syncSizeControls() {
  const enabled = !filling && !panning;

}

pad.addEventListener("pointerdown", (event) => {
  if (sketch.hidden) return;
  $("#sketch-err").textContent = "";
  if (filling) {
    const point = padPoint(event);
    fillAt(point.x, point.y);
    return;
  }
  pad.setPointerCapture(event.pointerId);
  if (panning) {
    panStart = {
      x: event.clientX,
      y: event.clientY,
      left: sketchViewport.scrollLeft,
      top: sketchViewport.scrollTop,
    };
    sketchViewport.classList.add("dragging");
    return;
  }
  currentStroke = {
    color: erasing ? PAPER : penColor,
    size: erasing ? Math.max(penSize * 4, 16) : penSize,
    points: [padPoint(event)],
  };
  paintSketch();
});
pad.addEventListener("pointermove", (event) => {
  if (!pad.hasPointerCapture(event.pointerId)) return;
  if (panning && panStart) {
    sketchViewport.scrollLeft = panStart.left - (event.clientX - panStart.x);
    sketchViewport.scrollTop = panStart.top - (event.clientY - panStart.y);
    return;
  }
  if (!currentStroke) return;
  currentStroke.points.push(padPoint(event));
  paintSketch();
});
function endStroke(event) {
  if (event && pad.hasPointerCapture(event.pointerId)) pad.releasePointerCapture(event.pointerId);
  if (panning) {
    panStart = null;
    sketchViewport.classList.remove("dragging");
    return;
  }
  const stroke = currentStroke;
  currentStroke = null;
  if (stroke?.points.length) {
    commit({ type: "stroke", stroke });
    renderBase();
  }
  paintSketch();
}
pad.addEventListener("pointerup", endStroke);
pad.addEventListener("pointercancel", endStroke);

document.addEventListener("keydown", (event) => {
  if (sketch.hidden) return;
  const key = event.key.toLowerCase();
  if ((event.ctrlKey || event.metaKey) && key === "z") {
    event.preventDefault();
    if (event.shiftKey) redoSketch();
    else undoSketch();
    return;
  }
  if ((event.ctrlKey || event.metaKey) && key === "y") {
    event.preventDefault();
    redoSketch();
    return;
  }
  if (event.ctrlKey || event.metaKey || event.altKey) return;
  if (key === "b") {
    setPenMode();
  } else if (key === "e") {
    setEraserMode();
  } else if (key === "f") {
    setFillMode();
  } else if (event.key === "[") {
    penSize = Math.max(2, penSize - 1);
  } else if (event.key === "]") {
    penSize = Math.min(30, penSize + 1);
  }
});

async function loadOlderMessages() {
  const id = selected, version = threadViewVersion;
  if (!id || !messagePages.hasOlder || messagePages.pending) return;
  try {
    const data = await messagePages.load(`${version}:${id}`, `/api/admin/threads/${id}`, {}, "older");
    if (id !== selected || version !== threadViewVersion) return;
    const scroller = messageScroller();
    const top = scroller.scrollTop, height = scroller.scrollHeight;
    renderThread(data);
    scroller.scrollTop = top + scroller.scrollHeight - height;
  } catch { /* The shared pager exposes a recoverable error beside the button. */ }
}
