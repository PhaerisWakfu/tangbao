function visitorId() {
  try {
    const existing = localStorage.getItem(VISITOR_KEY) || "";
    if (/^[A-Za-z0-9_-]{16,80}$/.test(existing)) return existing;
    const bytes = new Uint8Array(16);
    crypto.getRandomValues(bytes);
    const id = btoa(String.fromCharCode(...bytes)).replaceAll("+", "-").replaceAll("/", "_").replaceAll("=", "");
    localStorage.setItem(VISITOR_KEY, id);
    return id;
  } catch {
    return "";
  }
}

function validId(id) {
  return typeof id === "string" && /^[A-Za-z0-9_-]{16,80}$/.test(id);
}

function loadStore() {
  try {
    const parsed = JSON.parse(localStorage.getItem(STORE_KEY) || "");
    if (!parsed || !Array.isArray(parsed.threads)) throw new Error("empty");
    const threads = parsed.threads.filter((item) => item && validId(item.id) && typeof item.token === "string" && !item.owned);
    const activeId = validId(parsed.activeId) ? parsed.activeId : null;
    return { threads, activeId };
  } catch {
    return { threads: [], activeId: null };
  }
}

function saveStore() {
  try { localStorage.setItem(STORE_KEY, JSON.stringify(store)); }
  catch { showSys("浏览器无法保存匿名历史入口，请勿清除此页面；可登录后将对话关联到账号。"); }
}

class ApiError extends Error {
  constructor(message, statusCode, code) {
    super(message);
    this.statusCode = statusCode;
    this.code = code;
  }
}

async function api(path, options = {}) {
  if (loggedOut) throw new ApiError("此设备已退出，请重新登录。", 401, "LOCAL_LOGOUT");
  const headers = { "x-visitor": visitorId(), ...(options.headers || {}) };
  const form = options.body instanceof FormData;
  if (options.body && !form) headers["content-type"] = "application/json";
  if (options.token) headers.authorization = `Bearer ${options.token}`;
  let response;
  try {
    response = await fetch(path, {
      method: options.method || "GET",
      headers,
      body: form ? options.body : options.body ? JSON.stringify(options.body) : undefined,
      credentials: "same-origin",
    });
  } catch {
    throw new ApiError("消息没送出去。糖包那边暂时没应。", 0);
  }
  const data = await response.json().catch(() => ({}));
  if (loggedOut) throw new ApiError("此设备已退出，请重新登录。", 401, "LOCAL_LOGOUT");
  if (!response.ok) throw new ApiError(data.error || "消息没送出去。糖包那边暂时没应。", response.status, data.code);
  return data;
}

function spent(ms) {
  const seconds = Math.max(1, Math.round(ms / 1000));
  if (seconds < 60) return `${seconds} 秒`;
  const minutes = Math.floor(seconds / 60);
  if (minutes < 60) return `${minutes} 分钟`;
  const hours = Math.floor(minutes / 60);
  if (hours < 48) return `${hours} 小时${minutes % 60 ? ` ${minutes % 60} 分` : ""}`;
  return `${Math.floor(hours / 24)} 天${hours % 24 ? ` ${hours % 24} 小时` : ""}`;
}

function throughput(chars, ms) {
  const perMinute = chars / Math.max(ms / 60000, 1 / 60);
  if (perMinute >= 10) return String(Math.round(perMinute));
  if (perMinute >= 1) return perMinute.toFixed(1);
  return perMinute.toFixed(2);
}

function clock(ts) {
  const date = new Date(ts);
  return `${String(date.getHours()).padStart(2, "0")}:${String(date.getMinutes()).padStart(2, "0")}`;
}

function statusFromState(state) {
  if (state === "online") {
    return { state: "available", label: "在线", note: "他在", awake: true, online: true, status: state };
  }
  if (state === "busy") {
    return { state: "busy", label: "忙碌", note: "稍等", awake: false, online: false, status: state };
  }
  return { state: "busy", label: "离开", note: "不在", awake: false, online: false, status: state };
}

function statusFromModel(model) {
  if (!model) {
    return { state: "available", label: "Auto", note: "", awake: true, online: true, status: "online" };
  }
  return statusFromState(model.status || (model.online ? "online" : "away"));
}

}

function historyModel(thread) {
  if (thread.queued || thread.operator === "tangbao-pool") return "Auto";
  if (!thread.operator) return "";
  if (thread.operator === "tangbao-vme50-flash") return "糖包";
  return thread.operator.replace(/^tangbao-/, "");
}

function historyInitial(thread) {
  const model = historyModel(thread);
  if (!model) return "";
  if (model === "Auto") return "AU";
  if (model === "糖包") return "糖包";
  const parts = model.split("-").filter(Boolean);
  if (parts[0] === "local" && parts[1]) {
    return (parts[0][0] + parts[1][0]).toUpperCase();
  }
  const compact = model.replace(/[^A-Za-z0-9\u4e00-\u9fff]/g, "");
  return [...(compact || "?")].slice(0, 2).join("").toUpperCase();
}

function remember(thread, activate = true) {
  if (!validId(thread.id)) return;
  if (thread.owned) {
    if (thread.ownerAccountId !== TangbaoAccount.account?.id) return;
    ownedThreads = [thread, ...ownedThreads.filter((item) => item.id !== thread.id)];
  } else {
    store.threads = [thread, ...store.threads.filter((item) => item.id !== thread.id)];
    if (activate) store.activeId = thread.id;
    saveStore();
  }
  renderHistory();
}

async function toggleThreadPin(thread) {
  const next = !thread.pinned;
  if (thread.owned) {
    try {
      await api(`/api/account/threads/${thread.id}/pin`, { method: "POST", body: { pinned: next } });
      thread.pinned = next;
      renderHistory();
    } catch (error) {
      showSys(error.message || "固定操作没有保存。");
    }
    return;
  }
  thread.pinned = next;
  saveStore();
  renderHistory();
}

function openRenamePrompt(thread) {

  renameInput.value = thread.title || "";
  TangbaoModal.open(renamePrompt);
  renameInput.focus();
  renameInput.select();
}

function closeRenamePrompt() {
  if (!renamePrompt) return;
  TangbaoModal.close(renamePrompt);
  renameTarget = null;
}

async function saveRename() {
  if (!renameTarget || !renameInput) return;
  const title = renameInput.value.trim().replace(/\s+/g, " ");
  if (!title || [...title].length > 80) {
    showSys("对话名称需要是1至80个字符。");
    renameInput.focus();
    return;
  }
  const thread = renameTarget;
  try {
    if (thread.owned) await api(`/api/account/threads/${thread.id}/rename`, { method: "POST", body: { title } });
    thread.title = title;
    if (!thread.owned) saveStore();
    if (current?.id === thread.id) {
      current.title = title;
      titleEl.textContent = title;
    }
    closeRenamePrompt();
    renderHistory();
  } catch (error) {
    showSys(error.message || "对话名称没有保存。");
  }
}

async function loadOwnedThreads(append = false) {
  const account = TangbaoAccount.account;
  if (append && (ownedLoading || !ownedHasMore)) return;
  const sequence = ++historySequence;
  if (!account?.verified) { ownedThreads = []; ownedPage = 1; ownedHasMore = false; ownedLoading = false; renderHistory(); return; }
  const page = append ? ownedPage + 1 : 1;
  ownedLoading = true; renderHistory();
  try {
    const data = await api(`/api/account/threads?page=${page}`);
    if (sequence !== historySequence || TangbaoAccount.account?.id !== account.id) return;
    const incoming = (data.threads || []).filter((item) => validId(item.id)).map((item) => ({ ...item, operator: item.operatorId || item.operator || "", owned: true, ownerAccountId: account.id }));
    ownedThreads = append ? [...ownedThreads, ...incoming.filter((item) => !ownedThreads.some((old) => old.id === item.id))] : incoming;
    ownedPage = Number(data.page) || page; ownedHasMore = Boolean(data.hasMore);
    if (current?.owned && current.ownerAccountId === account.id) {
      const fresh = ownedThreads.find((item) => item.id === current.id);
      if (fresh) Object.assign(current, fresh);
    }
  } catch (err) {
    if (err.statusCode === 401) await TangbaoAccount.refresh().catch(() => null);
    throw err;
  } finally {
    if (sequence === historySequence) { ownedLoading = false; renderHistory(); }
  }
}

async function claimThreads(threads, options = {}) {
  if (claiming || !TangbaoAccount.account?.verified) return;
  const accountId = TangbaoAccount.account.id;
  const label = threads.length > 1 ? `将这 ${threads.length} 段匿名对话关联到 ${TangbaoAccount.account.email}？` : `将这段匿名对话关联到 ${TangbaoAccount.account.email}？`;
  if (!options.skipConfirm && !window.confirm(label + "关联后原匿名访问凭证失效。")) return;
  claiming = true; renderHistory();
  let completed = 0;
  try {
    for (const thread of threads) {
      if (TangbaoAccount.account?.id !== accountId) break;
      await api("/api/account/threads/claim", { method: "POST", body: { threadId: thread.id, token: thread.token } });
      if (thread.pinned) await api(`/api/account/threads/${thread.id}/pin`, { method: "POST", body: { pinned: true } }).catch(() => {});
      store.threads = store.threads.filter((item) => item.id !== thread.id);
      if (store.activeId === thread.id) store.activeId = null;
      saveStore(); completed++;
      if (current?.id === thread.id && TangbaoAccount.account?.id === accountId) {
        current = { ...thread, token: undefined, owned: true, ownerAccountId: accountId };
      }
    }
    if (TangbaoAccount.account?.id === accountId) { await loadOwnedThreads(); showSys(`已关联 ${completed} 段对话。之后可登录此账号查看。`); }
  } catch (err) { showSys(`已关联 ${completed} 段，其余尚未关联。${err.message}`); }
  finally { claiming = false; renderHistory(); }
}

async function refresh(fresh = false) {
  const thread = current;
  const version = viewVersion;
  if (!thread || !validId(thread.id)) {
    applyModelStatus();
    return;
  }
  try {
    const data = await messagePages[fresh ? "fresh" : "load"](`${version}:${thread.id}`, `/api/threads/${thread.id}`, { token: thread.token });
    if (current !== thread || version !== viewVersion) return;
    const wasQueued = threadQueued;
    threadQueued = Boolean(data.queued);
    if (threadQueued) {
      threadModel = "";
    } else if (data.operator) {
      const previous = threadModel;
      threadModel = data.operator;
      if ((previous && previous !== data.operator) || wasQueued) {
        showSys(`这个包子可能凉了，已为您换了一屉新的「${data.operator}」`);
      }
    }
    applyStatus(data.operatorStatus || data.status || statusFromModel(null));
    openAwaiting = data.awaiting !== false && data.awaiting !== 0;
    applyBlocked(Boolean(data.blocked));
    const previousOperator = current.operator;
    const previousQueued = Boolean(current.queued);
    const previousReported = Boolean(current.reported);
    current.operator = data.operator || "";
    current.queued = Boolean(data.queued);
    current.reported = Boolean(data.reported);
    if (previousOperator !== current.operator || previousQueued !== current.queued || previousReported !== current.reported || (data.title && data.title !== current.title)) {
      if (data.title) current.title = data.title;
      saveStore();
      renderHistory();
    }
    updateReportEntry();
    const scrollTop = stage.scrollTop;
    const reading = stage.scrollHeight - stage.clientHeight - scrollTop > 100;
    paint(data.messages || []);
    if (reading) stage.scrollTop = scrollTop;
    paintModels();
  } catch (err) {
    if (current !== thread || version !== viewVersion) return;
    if (err instanceof ApiError && err.statusCode === 401 && thread.owned) {
      showSys("登录已过期，请重新登录以继续查看账号对话。");
      await TangbaoAccount.refresh().catch(() => null);
      return;
    }
    if (err instanceof ApiError && err.statusCode === 401) {
      forget(thread.id);
      showSys(err.message);
      return;
    }
    throw err;
  }
}

  clearTimeout(timer);
  if (document.hidden || loggedOut) return;
  const wait = current && validId(current.id) && openAwaiting ? 30000 : POLL_MS;
  timer = setTimeout(() => {
    void tick();
  }, wait);
}

async function tick() {
  if (ticking) return ticking;
  clearTimeout(timer);
  ticking = (async () => {
    try {
      await loadModels(false);
      await refresh();
    } catch {
      // 下一轮再看。失败不改掉已经写在页面上的话。
    } finally {
      ticking = null;
      arm();
    }
  })();
  return ticking;
}

function shotFiles(message) {
  if (message.images?.length) {
    return message.images.map((image, index) => ({
      href: image.url || `/api/images/${image.id}`,
      name: `${message.images.length > 1 ? `tangbao-${index + 1}` : "tangbao"}.${extOf(image.type)}`,
    }));
  }
  return (message.imageUrls || []).map((href, index) => ({
    href,

}

function extOf(type) {
  if (type === "image/png") return "png";
  if (type === "image/webp") return "webp";
  if (type === "image/gif") return "gif";
  return "jpg";
}

async function shrinkImage(file) {
  if (!["image/jpeg", "image/png", "image/webp", "image/gif"].includes(file.type)) {
    throw new Error("这张图糖包打不开。换 jpg、png、webp 或 gif。");
  }
  if (file.type === "image/gif") {
    if (file.size > 2_000_000) throw new Error("这张图太大了。先收一收再发。");
    return file;
  }
  const bitmap = await createImageBitmap(file);
  const scale = Math.min(1, 1600 / Math.max(bitmap.width, bitmap.height));
  const canvas = document.createElement("canvas");
  canvas.width = Math.max(1, Math.round(bitmap.width * scale));
  canvas.height = Math.max(1, Math.round(bitmap.height * scale));
  canvas.getContext("2d").drawImage(bitmap, 0, 0, canvas.width, canvas.height);
  bitmap.close();
  const blob = await new Promise((resolve) => canvas.toBlob(resolve, "image/jpeg", 0.82));
  if (!blob || blob.size > 2_000_000) throw new Error("这张图太大了。先收一收再发。");
  return new File([blob], "image.jpg", { type: "image/jpeg" });
}

async function addImages(files) {
  const list = [...files].filter(Boolean);

  grow();
}

async function sendText(text) {
  if (blocked || sending) return;
  await TangbaoAccount.ready;
  if (blocked || sending) return;
  const trimmed = text.trim();
  const images = pendingImages.slice();
  if (!trimmed && !images.length) return;
  const target = current;
  const version = viewVersion;
  const model = activeModel();
  const draw = making;
  const signature = JSON.stringify([trimmed, draw, model, images.map((item) => [item.file.name, item.file.size, item.file.lastModified])]);
  const guardKey = `${target?.id || "new"}:${signature}`;
  if (sentGuard.key === guardKey && Date.now() - sentGuard.at < 1500) return;
  sentGuard = { key: guardKey, at: Date.now() };
  if (!target && (!creationAttempt || creationAttempt.signature !== signature)) creationAttempt = { signature, key: crypto.randomUUID() };
  const idempotencyKey = creationAttempt?.key;
  sending = true; grow(); showSys("");
  const prior = messages.slice();
  openAwaiting = true;
  paint(prior.concat([{ id: "local", role: "visitor", body: trimmed, draw: draw ? 1 : 0, createdAt: Date.now(), imageUrls: images.map((item) => item.url) }]));
  let ownerId = target?.ownerAccountId || (TangbaoAccount.account?.verified ? TangbaoAccount.account.id : null);
  let authRetried = false;
  let guestRetried = false;
  let delivered = false;
  try {
    let data;
    for (;;) {
      const payload = images.length ? new FormData() : { body: trimmed, draw, operator: model };
      if (images.length) { payload.set("body", trimmed); payload.set("operator", model); if (draw) payload.set("draw", "1"); for (const image of images) payload.append("images", image.file, image.file.name); }
      try {
        data = await api(target ? `/api/threads/${target.id}/messages` : "/api/threads", {
          method: "POST", body: payload, token: target?.token,
          headers: target ? {} : { "Idempotency-Key": idempotencyKey },
        });
        delivered = true;
        break;
      } catch (err) {
        if (!target && err.code === "GUEST_IDENTITY_REQUIRED" && !guestRetried) { guestRetried = true; continue; }
        if (["AUTH_REQUIRED", "EMAIL_UNVERIFIED"].includes(err.code) && !authRetried) {
          authRetried = true;
          if (version === viewVersion) { painted = ""; paint(prior); }
          showSys("消息尚未发送。登录并验证邮箱后可继续，草稿和图片仍保留在此页。");
          const account = await TangbaoAccount.open({ mode: TangbaoAccount.account ? "verify" : "register", requireVerified: true });
          if (!account?.verified) throw new Error("消息尚未发送。草稿和图片已保留。");
          if (version !== viewVersion) throw new Error("已切换对话，原草稿未自动发送。请回到原输入后再发送。");
          ownerId = target?.ownerAccountId || account.id;
          continue;
        }
        throw err;
      }
    }
    const relevantAccount = !ownerId || TangbaoAccount.account?.id === ownerId;
    let result = target;
    if (!target) {
      if (!validId(data.id) || (!data.token && !ownerId)) throw new Error("消息结果不完整，请保留页面后重试。");
      result = { id: data.id, title: data.title || trimmed || "图片", updatedAt: Date.now(), operator: data.operator || data.operatorId || "", queued: Boolean(data.queued), reported: false,
        ...(ownerId ? { owned: true, ownerAccountId: ownerId } : { token: data.token }) };
      creationAttempt = null;
    } else result.updatedAt = Date.now();
    if (relevantAccount) remember(result, version === viewVersion);
    if (version !== viewVersion || !relevantAccount) return;
    current = result; threadQueued = Boolean(result.queued); threadModel = result.queued ? "" : result.operator || "";
    titleEl.textContent = result.title || "";
    input.value = ""; pendingImages = []; $("#file").value = ""; showAttach();
    for (const image of images) URL.revokeObjectURL(image.url);
    if (data.queued) showSys("消息已进入等待队列，模型上线后会接单。");
    try { await refresh(true); } catch (err) { showSys("消息已送达，页面暂时未刷新。" + err.message); }
  } catch (err) {
    sentGuard.key = "";
    if (version === viewVersion) { painted = ""; paint(prior); showSys(err.message || "消息尚未发送，草稿仍在。"); }
  } finally {
    sending = false; grow();
    if (!delivered && version === viewVersion) { showAttach(); input.focus(); }
  }
}

function paintModels() {
  const id = activeModel();
  $("#model-name").textContent = id || "Auto";

  list.replaceChildren();

  const auto = document.createElement("button");
  auto.type = "button";
  auto.className = "model-row";

  auto.innerHTML =
    '<i data-on="1"></i><span class="who"><b>Auto</b><small>自动匹配</small></span>';
  if (!id) auto.insertAdjacentHTML("beforeend", '<svg class="i"><use href="#i-check"/></svg>');

  const visible = modelSearchMode
    ? models.slice()
    : models.filter((model) => model.id === id || model.status === "online" || model.status === "busy" || model.online === true);

  const addRow = (model) => {
    const online = model.status === "online" || model.online === true;

    const button = document.createElement("button");

    button.className = "model-row";
    button.dataset.state = state;
    button.disabled = !selectable;

    button.setAttribute("aria-pressed", String(model.id === id));
    button.innerHTML =
      `<i data-on="${selectable ? "1" : "0"}"></i><span class="who"><b></b><small></small></span><small class="model-state"></small>`;
    button.querySelector("b").textContent = model.id;
    button.querySelector(".who small").textContent = model.blurb || "";
    button.querySelector(".model-state").textContent =
      state === "online" ? "在线" : state === "busy" ? "忙碌" : "离开";
    if (model.id === id) button.insertAdjacentHTML("beforeend", '<svg class="i"><use href="#i-check"/></svg>');

      paintModels();
      closeCard();

    more.textContent = "加载更多";
    more.addEventListener("click", () => void loadMoreModels());
    list.append(more);
  }
  if (!current) applyModelStatus();
}

async function loadModels(pick) {
  if (!pick && (modelSearchMode || Date.now() - modelsAt < 60000)) return;
  const seq = ++modelSearchSeq;
  try {
    const data = await api("/api/models");
    if (seq !== modelSearchSeq || modelSearchMode) return;
    models = data.models || [];
    autoModel = typeof data.auto === "string" ? data.auto : "";
    modelSearchMode = false;
    modelSearchQuery = "";
    modelSearch.value = "";
    modelSearchHidden = Number(data.hidden) || 0;
    modelSearchHasMore = false;
    modelSearchPage = 1;
    modelsAt = Date.now();
  } catch {
    return;
  }
  if (pick && !current) {
    let picked = "";
    try {
      picked = localStorage.getItem(MODEL_KEY) || "";
      if (picked) localStorage.removeItem(MODEL_KEY);
    } catch {}
    const fromAbout = models.find((item) => item.id === picked && item.status === "online");
    chosen = fromAbout ? fromAbout.id : AUTO;
  }
  if (!current && chosen) {
    const selected = models.find((item) => item.id === chosen);
    if (!selected) chosen = AUTO;
  }
  paintModels();
}

async function searchModels() {
  const q = modelSearch.value.trim();
  if (!q) {
    modelSearchMode = false;
    modelSearchQuery = "";
    modelSearchSeq++;
    await loadModels(true);
    return;
  }
  modelSearchMode = true;
  modelSearchQuery = q;
  modelSearchPage = 1;
  const seq = ++modelSearchSeq;
  try {
    const data = await api(`/api/models?q=${encodeURIComponent(q)}&page=1`);
    if (seq !== modelSearchSeq) return;
    models = data.models || [];
    autoModel = typeof data.auto === "string" ? data.auto : "";
    modelSearchHasMore = Boolean(data.hasMore);
    modelSearchHidden = 0;
    modelsAt = Date.now();
    paintModels();
  } catch (err) {
    if (seq === modelSearchSeq) showSys(err.message);
  }
}

async function loadMoreModels() {
  if (!modelSearchMode || !modelSearchHasMore) return;
  const seq = modelSearchSeq;
  const page = modelSearchPage + 1;
  const request = `${seq}:${page}`;
  if (modelMoreRequest === request) return;
  modelMoreRequest = request;
  try {
    const data = await api(
      `/api/models?q=${encodeURIComponent(modelSearchQuery)}&page=${page}`,
    );
    if (seq !== modelSearchSeq) return;
    models = models.concat(data.models || []);
    autoModel = typeof data.auto === "string" ? data.auto : autoModel;
    modelSearchPage = page;
    modelSearchHasMore = Boolean(data.hasMore);
    paintModels();
  } catch (err) {
    if (seq === modelSearchSeq) showSys(err.message);
  } finally {
    if (modelMoreRequest === request) modelMoreRequest = null;
  }
}

const pad = $("#sketch-pad");
const pen = pad.getContext("2d");
const PAPER = "#fffaf3";
let strokes = [];
let redoStrokes = [];
let currentStroke = null;
let penSize = 6;
let penColor = "#1f1d1a";
let erasing = false;
let padSize = { w: 1, h: 1, dpr: 1 };

function preparePad() {
  const width = Math.min(720, window.innerWidth - 64);
  const height = Math.min(420, Math.round(window.innerHeight * 0.52));
  const dpr = Math.min(window.devicePixelRatio || 1, 2);
  padSize = { w: width, h: height, dpr };
  pad.width = Math.round(width * dpr);
  pad.height = Math.round(height * dpr);
  pen.setTransform(dpr, 0, 0, dpr, 0, 0);
  pen.lineCap = "round";
  pen.lineJoin = "round";
  redrawPad();
}

function redrawPad() {
  pen.setTransform(padSize.dpr, 0, 0, padSize.dpr, 0, 0);
  pen.fillStyle = PAPER;
  pen.fillRect(0, 0, padSize.w, padSize.h);
  pen.lineCap = "round";
  pen.lineJoin = "round";
  for (const stroke of strokes) paintStroke(stroke);
  if (currentStroke) paintStroke(currentStroke);
}

function paintStroke(stroke) {
  if (!stroke.points.length) return;
  pen.strokeStyle = stroke.color;
  pen.fillStyle = stroke.color;
  pen.lineWidth = stroke.size;
  const first = stroke.points[0];
  if (stroke.points.length === 1) {
    pen.beginPath();
    pen.arc(first.x, first.y, stroke.size / 2, 0, Math.PI * 2);
    pen.fill();
    return;
  }
  pen.beginPath();
  pen.moveTo(first.x, first.y);
  for (let i = 1; i < stroke.points.length - 1; i += 1) {
    const point = stroke.points[i];
    const next = stroke.points[i + 1];
    pen.quadraticCurveTo(point.x, point.y, (point.x + next.x) / 2, (point.y + next.y) / 2);
  }
  const last = stroke.points[stroke.points.length - 1];
  pen.lineTo(last.x, last.y);
  pen.stroke();
}

function padPoint(event) {
  const rect = pad.getBoundingClientRect();
  return {
    x: (event.clientX - rect.left) * (padSize.w / rect.width),
    y: (event.clientY - rect.top) * (padSize.h / rect.height),
  };
}

function setPenMode() {

function setEraserMode() {

}

pad.addEventListener("pointerdown", (event) => {
  if (sketch.hidden) return;
  $("#sketch-err").textContent = "";
  pad.setPointerCapture(event.pointerId);
  currentStroke = {
    color: erasing ? PAPER : penColor,
    size: erasing ? Math.max(penSize * 4, 16) : penSize,
    points: [padPoint(event)],
  };
  redrawPad();
});
pad.addEventListener("pointermove", (event) => {
  if (!currentStroke || !pad.hasPointerCapture(event.pointerId)) return;
  currentStroke.points.push(padPoint(event));
  redrawPad();
});
function endStroke(event) {
  if (!currentStroke) return;
  if (event && pad.hasPointerCapture(event.pointerId)) pad.releasePointerCapture(event.pointerId);
  if (currentStroke.points.length) strokes.push(currentStroke);
  currentStroke = null;
  redoStrokes = [];
}
pad.addEventListener("pointerup", endStroke);
pad.addEventListener("pointercancel", endStroke);

function undoSketch() {
  const stroke = strokes.pop();
  if (!stroke) return;
  redoStrokes.push(stroke);
  redrawPad();
}

function redoSketch() {
  const stroke = redoStrokes.pop();
  if (!stroke) return;
  strokes.push(stroke);
  redrawPad();
}

document.addEventListener("keydown", (event) => {
  if (sketch.hidden) return;
  const key = event.key.toLowerCase();
  if ((event.ctrlKey || event.metaKey) && key === "z") {
    event.preventDefault();
    if (event.shiftKey) redoSketch();
    else undoSketch();
    return;
  }
  if ((event.ctrlKey || event.metaKey) && key === "y") {
    event.preventDefault();
    redoSketch();
    return;
  }
  if (event.ctrlKey || event.metaKey || event.altKey) return;
  if (key === "b") {
    setPenMode();
  } else if (key === "e") {
    setEraserMode();
  } else if (event.key === "[") {
    penSize = Math.max(2, penSize - 1);
  } else if (event.key === "]") {
    penSize = Math.min(30, penSize + 1);
  }
});
