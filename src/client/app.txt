function visitorId() {
  try {
    const existing = localStorage.getItem(VISITOR_KEY) || "";
    if (/^[A-Za-z0-9_-]{16,80}$/.test(existing)) return existing;
    const bytes = new Uint8Array(16);
    crypto.getRandomValues(bytes);
    const id = btoa(String.fromCharCode(...bytes)).replaceAll("+", "-").replaceAll("/", "_").replaceAll("=", "");
    localStorage.setItem(VISITOR_KEY, id);
    return id;
  } catch {
    return "";
  }
}

function validId(id) {
  return typeof id === "string" && /^[A-Za-z0-9_-]{16,80}$/.test(id);
}

function loadStore() {
  try {
    const parsed = JSON.parse(localStorage.getItem(STORE_KEY) || "");
    if (!parsed || !Array.isArray(parsed.threads)) throw new Error("empty");
    const threads = parsed.threads.filter((item) => item && validId(item.id));
    const activeId = validId(parsed.activeId) ? parsed.activeId : null;
    return { threads, activeId };
  } catch {
    return { threads: [], activeId: null };
  }
}

function saveStore() {
  localStorage.setItem(STORE_KEY, JSON.stringify(store));
}

class ApiError extends Error {
  constructor(message, statusCode) {
    super(message);
    this.statusCode = statusCode;
  }
}

async function api(path, options = {}) {
  const headers = { "x-visitor": visitorId() };
  const form = options.body instanceof FormData;
  if (options.body && !form) headers["content-type"] = "application/json";
  if (options.token) headers.authorization = `Bearer ${options.token}`;
  let response;
  try {
    response = await fetch(path, {
      method: options.method || "GET",
      headers,
      body: form ? options.body : options.body ? JSON.stringify(options.body) : undefined,
      credentials: "same-origin",
    });
  } catch {
    throw new ApiError("消息没送出去。糖包那边暂时没应。", 0);
  }
  const data = await response.json().catch(() => ({}));
  if (!response.ok) throw new ApiError(data.error || "消息没送出去。糖包那边暂时没应。", response.status);
  return data;
}

function spent(ms) {
  const seconds = Math.max(1, Math.round(ms / 1000));
  if (seconds < 60) return `${seconds} 秒`;
  const minutes = Math.floor(seconds / 60);
  if (minutes < 60) return `${minutes} 分钟`;
  const hours = Math.floor(minutes / 60);
  if (hours < 48) return `${hours} 小时${minutes % 60 ? ` ${minutes % 60} 分` : ""}`;
  return `${Math.floor(hours / 24)} 天${hours % 24 ? ` ${hours % 24} 小时` : ""}`;
}

function throughput(chars, ms) {
  const perMinute = chars / Math.max(ms / 60000, 1 / 60);
  if (perMinute >= 10) return String(Math.round(perMinute));
  if (perMinute >= 1) return perMinute.toFixed(1);
  return perMinute.toFixed(2);
}

function clock(ts) {
  const date = new Date(ts);
  return `${String(date.getHours()).padStart(2, "0")}:${String(date.getMinutes()).padStart(2, "0")}`;
}

function applyStatus(next) {
  const present = next.status === "online" || next.status === "busy" || Boolean(next.awake);
  status = { ...next, awake: present, online: next.status === "online" };
  body.dataset.presence = present ? "awake" : "asleep";
  try {
    localStorage.setItem("tangbao.presence", body.dataset.presence);
  } catch {}
  if (!body.classList.contains("settled")) requestAnimationFrame(() => body.classList.add("settled"));
  $("#status-label").textContent = next.label;
  $("#status-note").textContent = next.note;
  $("#status").title = present ? "他在，消息会被看到" : "他不在，消息会留到他回来";
  tickPending();
}

function statusFromState(state) {
  if (state === "online") {
    return { state: "available", label: "在线", note: "他在", awake: true, online: true, status: state };
  }
  if (state === "busy") {
    return { state: "busy", label: "忙碌", note: "稍等", awake: false, online: false, status: state };
  }
  return { state: "busy", label: "离开", note: "不在", awake: false, online: false, status: state };
}

function statusFromModel(model) {
  if (!model) {
    return { state: "available", label: "自动匹配", note: "在线模型优先", awake: true, online: true, status: "online" };
  }
  return statusFromState(model.status || (model.online ? "online" : "away"));
}

function applyModelStatus() {
  const id = activeModel();
  if (!id) {
    applyStatus(statusFromModel(null));
    return;
  }
  applyStatus(statusFromModel(models.find((item) => item.id === id)));
}

function botRow(message, askedAt) {
  const took = Math.max(0, message.createdAt - askedAt);
  const row = document.createElement("article");
  row.className = "b";
  row.innerHTML =
    '<svg class="avatar"><use href="#pleat"/></svg>' +
    '<div class="b-body">' +
    '<details class="thought"><summary></summary>' +
    '<div class="thought-body"><dl></dl><p>思考过程由一个人完成，没有留下草稿。</p></div></details>' +
    '<p class="b-text"></p>' +
    '<div class="acts">' +
    '<button type="button" class="copy" data-tip="复制" aria-label="复制"><svg class="i"><use href="#i-copy"/></svg></button>' +
    '<button type="button" aria-disabled="true" data-tip="他只写一遍" aria-label="重新生成"><svg class="i"><use href="#i-redo"/></svg></button>' +
    "<time></time>" +
    "</div></div>";
  const summary = row.querySelector("summary");
  summary.textContent = `已思考（用时 ${spent(took)}）`;
  summary.insertAdjacentHTML("beforeend", '<svg class="i"><use href="#i-chev"/></svg>');
  const facts = [
    ["推理核心", "1 个（糖包本人）"],
    ["用时", spent(took)],
    ["吞吐", `约 ${throughput([...message.body].length, took)} 字/分钟`],
  ];
  const list = row.querySelector("dl");
  for (const [term, value] of facts) {
    const dt = document.createElement("dt");
    dt.textContent = term;
    const dd = document.createElement("dd");
    dd.textContent = value;
    list.append(dt, dd);
  }
  const text = row.querySelector(".b-text");
  if (message.body) {
    text.classList.add("md");
    text.innerHTML = typeof renderMarkdown === "function" ? renderMarkdown(message.body) : "";
    if (typeof renderMarkdown !== "function") text.textContent = message.body;
  } else text.remove();
  appendShot(row.querySelector(".b-body"), message);
  const files = shotFiles(message);
  if (files.length) {
    const save = document.createElement("button");
    save.type = "button";
    save.dataset.tip = "下载图片";
    save.setAttribute("aria-label", "下载图片");
    save.innerHTML = '<svg class="i"><use href="#i-down"/></svg>';
    save.addEventListener("click", () => downloadShots(files));
    row.querySelector('[aria-label="重新生成"]').after(save);
  }
  const time = row.querySelector("time");
  time.textContent = clock(message.createdAt);
  time.dateTime = new Date(message.createdAt).toISOString();
  return row;
}

function tickPending() {
  const row = log.querySelector(".pending");
  if (!row) return;
  const wait = status.awake ? WAIT.awake : WAIT.asleep;
  row.querySelector("small").textContent = `已用时 ${spent(Date.now() - Number(row.dataset.since))} · ${wait}`;
}

function paint(list) {
  const key = JSON.stringify({ list, openAwaiting, blocked });
  if (key === painted) {
    tickPending();
    return;
  }
  painted = key;
  messages = list;
  body.dataset.state = list.length ? "chat" : "empty";
  titleEl.textContent = list.length && current ? current.title || "" : "";
  log.replaceChildren();
  let askedAt = 0;
  for (const message of list) {
    if (message.role === "visitor") {
      if (!askedAt) askedAt = message.createdAt;
      const bubble = document.createElement("div");
      bubble.className = "u";
      if (message.draw) {
        const tag = document.createElement("span");
        tag.className = "u-make";
        tag.textContent = "制作图片";
        bubble.append(tag);
      }
      if (message.body) bubble.append(document.createTextNode(message.body));
      appendShot(bubble, message);
      log.append(bubble);
    } else {
      log.append(botRow(message, askedAt || message.createdAt));
      askedAt = 0;
    }
  }
  if (askedAt && !blocked && openAwaiting) log.append(pendingRow(askedAt));
  else if (askedAt && !blocked) {
    const note = document.createElement("p");
    note.className = "letgo";
    note.textContent = "他先把这句放下了。";
    log.append(note);
  }
  tickPending();
  stage.scrollTop = stage.scrollHeight;
}

function renderHistory() {
  history.replaceChildren();
  if (!store.threads.length) {
    const empty = document.createElement("p");
    empty.className = "hist-empty";
    empty.textContent = "还没有对话。问的第一句会出现在这里。";
    history.append(empty);
    return;
  }
  const today = new Date().toDateString();
  const groups = new Map();
  for (const thread of store.threads) {
    const label = new Date(thread.updatedAt || Date.now()).toDateString() === today ? "今天" : "更早";
    if (!groups.has(label)) groups.set(label, []);
    groups.get(label).push(thread);
  }
  for (const [label, threads] of groups) {
    const heading = document.createElement("h3");
    heading.textContent = label;
    history.append(heading);
    for (const thread of threads) {
      const row = document.createElement("div");
      row.className = "hist-row";
      if (current && current.id === thread.id) row.dataset.current = "true";
      const link = document.createElement("a");
      link.href = "/";
      link.textContent = thread.title || "新对话";
      link.addEventListener("click", (event) => {
        event.preventDefault();
        void openThread(thread.id);
      });
      const remove = document.createElement("button");
      remove.type = "button";
      remove.className = "hist-x";
      remove.setAttribute("aria-label", "删除这段对话");
      const icon = '<svg class="i"><use href="#i-x"/></svg>';
      remove.innerHTML = icon;
      remove.addEventListener("click", (event) => {
        event.preventDefault();
        event.stopPropagation();
        if (remove.dataset.confirm !== "1") {
          remove.dataset.confirm = "1";
          remove.setAttribute("aria-label", "确认删除");
          remove.textContent = "删";
          window.setTimeout(() => {
            if (remove.dataset.confirm !== "1") return;
            remove.dataset.confirm = "";
            remove.setAttribute("aria-label", "删除这段对话");
            remove.innerHTML = icon;
          }, 2200);
          return;
        }
        void removeThread(thread);
      });
      row.append(link, remove);
      history.append(row);
    }
  }
}

function remember(thread) {
  if (!validId(thread.id)) return;
  store.threads = [thread, ...store.threads.filter((item) => item.id !== thread.id)];
  store.activeId = thread.id;
  saveStore();
  renderHistory();
}

async function removeThread(thread) {
  try {
    await api(`/api/threads/${thread.id}`, { method: "DELETE", token: thread.token });
  } catch (err) {
    showSys(err.message || "这段对话还在。过一会儿再删。");
    return;
  }
  forget(thread.id);
  showSys("");
}

function forget(id) {
  store.threads = store.threads.filter((item) => item.id !== id);
  if (store.activeId === id) store.activeId = null;
  saveStore();
  current = null;
  painted = "";
  openAwaiting = true;
  paint([]);
  renderHistory();
}

async function refresh() {
  if (!current || !validId(current.id)) {
    applyModelStatus();
    return;
  }
  try {
    const data = await api(`/api/threads/${current.id}`, { token: current.token });
    if (data.operator) threadModel = data.operator;
    applyStatus(data.operatorStatus || data.status || statusFromModel(null));
    openAwaiting = data.awaiting !== false && data.awaiting !== 0;
    applyBlocked(Boolean(data.blocked));
    if (data.title && data.title !== current.title) {
      current.title = data.title;
      saveStore();
      renderHistory();
    }
    paint(data.messages || []);
    paintModels();
  } catch (err) {
    if (err instanceof ApiError && err.statusCode === 401) {
      forget(current.id);
      showSys(err.message);
      return;
    }
    throw err;
  }
}

function arm() {
  clearTimeout(timer);
  if (document.hidden) return;
  const wait = current && validId(current.id) && openAwaiting ? 3000 : POLL_MS;
  timer = setTimeout(() => {
    void tick();
  }, wait);
}

async function tick() {
  try {
    await loadModels(false);
    await refresh();
  } catch {
    // 下一轮再看。失败不改掉已经写在页面上的话。
  }
  arm();
}

async function openThread(id) {
  const found = store.threads.find((item) => item.id === id);
  if (!found) return;
  current = found;
  threadModel = "";
  store.activeId = id;
  saveStore();
  renderHistory();
  closeSide();
  showSys("");
  painted = "";
  log.replaceChildren();
  body.dataset.state = "chat";
  titleEl.textContent = found.title || "";
  try {
    await refresh();
  } catch (err) {
    showSys(err.message);
  }
}

function shotFiles(message) {
  if (message.images?.length) {
    return message.images.map((image, index) => ({
      href: `/api/images/${image.id}`,
      name: `${message.images.length > 1 ? `tangbao-${index + 1}` : "tangbao"}.${extOf(image.type)}`,
    }));
  }
  return (message.imageUrls || []).map((href, index) => ({
    href,
    name: `tangbao-${index + 1}.jpg`,
  }));
}

function extOf(type) {
  if (type === "image/png") return "png";
  if (type === "image/webp") return "webp";
  if (type === "image/gif") return "gif";
  return "jpg";
}

function downloadShots(files) {
  for (const file of files) {
    const link = document.createElement("a");
    link.href = file.href;
    link.download = file.name;
    document.body.append(link);
    link.click();
    link.remove();
  }
}

function appendShot(parent, message) {
  const sources = shotFiles(message);
  if (!sources.length) return;
  const row = document.createElement("div");
  row.className = "shots";
  for (const file of sources) {
    const img = document.createElement("img");
    img.className = "shot";
    img.alt = "图片，点击放大";
    img.src = file.href;
    img.tabIndex = 0;
    img.setAttribute("role", "button");
    const show = () => openViewer(file.href);
    img.addEventListener("click", show);
    img.addEventListener("keydown", (event) => {
      if (event.key !== "Enter" && event.key !== " ") return;
      event.preventDefault();
      show();
    });
    row.append(img);
  }
  const acts = parent.querySelector(".acts");
  if (acts) parent.insertBefore(row, acts);
  else parent.append(row);
}

async function shrinkImage(file) {
  if (!["image/jpeg", "image/png", "image/webp", "image/gif"].includes(file.type)) {
    throw new Error("这张图糖包打不开。换 jpg、png、webp 或 gif。");
  }
  if (file.type === "image/gif") {
    if (file.size > 2_000_000) throw new Error("这张图太大了。先收一收再发。");
    return file;
  }
  const bitmap = await createImageBitmap(file);
  const scale = Math.min(1, 1600 / Math.max(bitmap.width, bitmap.height));
  const canvas = document.createElement("canvas");
  canvas.width = Math.max(1, Math.round(bitmap.width * scale));
  canvas.height = Math.max(1, Math.round(bitmap.height * scale));
  canvas.getContext("2d").drawImage(bitmap, 0, 0, canvas.width, canvas.height);
  bitmap.close();
  const blob = await new Promise((resolve) => canvas.toBlob(resolve, "image/jpeg", 0.82));
  if (!blob || blob.size > 2_000_000) throw new Error("这张图太大了。先收一收再发。");
  return new File([blob], "image.jpg", { type: "image/jpeg" });
}

function showAttach() {
  const box = $("#attach");
  box.replaceChildren();
  if (!pendingImages.length) {
    box.hidden = true;
    return;
  }
  pendingImages.forEach((item, index) => {
    const thumb = document.createElement("div");
    thumb.className = "thumb";
    const img = document.createElement("img");
    img.src = item.url;
    img.alt = "";
    const remove = document.createElement("button");
    remove.type = "button";
    remove.textContent = "×";
    remove.setAttribute("aria-label", "拿掉这张图");
    remove.addEventListener("click", () => removeImage(index));
    thumb.append(img, remove);
    box.append(thumb);
  });
  box.hidden = false;
}

function removeImage(index) {
  const [gone] = pendingImages.splice(index, 1);
  if (gone) URL.revokeObjectURL(gone.url);
  showAttach();
  grow();
}

function clearAttach() {
  for (const item of pendingImages) URL.revokeObjectURL(item.url);
  pendingImages = [];
  $("#file").value = "";
  showAttach();
  grow();
}

async function addImages(files) {
  const list = [...files].filter(Boolean);
  if (!list.length) return;
  if (pendingImages.length + list.length > MAX_IMAGES) {
    showSys(`一次最多 ${MAX_IMAGES} 张。`);
  }
  const room = MAX_IMAGES - pendingImages.length;
  for (const file of list.slice(0, room)) {
    const shrunk = await shrinkImage(file);
    pendingImages.push({ file: shrunk, url: URL.createObjectURL(shrunk) });
  }
  showAttach();
  grow();
}

function grow() {
  input.style.height = "auto";
  input.style.height = `${Math.min(input.scrollHeight, 220)}px`;
  send.disabled = blocked || sending || (making ? !input.value.trim() : !input.value.trim() && !pendingImages.length);
}

async function sendText(text) {
  if (blocked || sending) return;
  const trimmed = text.trim();
  const images = pendingImages.slice();
  if (!trimmed && !images.length) return;
  const key = `${trimmed}\n${images.map((item) => `${item.file.size}:${item.file.lastModified}`).join(",")}`;
  const now = Date.now();
  if (sentGuard.key === key && now - sentGuard.at < 1500) return;
  sending = true;
  sentGuard = { key, at: now };
  send.disabled = true;
  showSys("");
  const prior = messages.slice();
  openAwaiting = true;
  paint(prior.concat([{
    id: "local",
    role: "visitor",
    body: trimmed,
    draw: making ? 1 : 0,
    createdAt: Date.now(),
    imageUrls: images.map((item) => item.url),
  }]));
  if (!current) titleEl.textContent = trimmed || (images.length > 1 ? `${images.length} 张图片` : "一张图片");
  input.value = "";
  pendingImages = [];
  $("#file").value = "";
  showAttach();
  grow();
  const payload = images.length ? new FormData() : { body: trimmed, draw: making };
  if (images.length) {
    payload.set("body", trimmed);
    if (making) payload.set("draw", "1");
    for (const item of images) payload.append("images", item.file, item.file.name);
  }
  try {
  if (!current || !validId(current.id)) {
    const created = await api("/api/threads", { method: "POST", body: withOperator(payload) });
    if (!validId(created.id) || !created.token) throw new ApiError("消息没送出去。糖包那边暂时没应。", 500);
    current = { id: created.id, token: created.token, title: created.title, updatedAt: Date.now() };
    if (created.operator) threadModel = created.operator;
    if (created.fallback && created.requested && created.operator) {
      showSys(`他刚走，这次由 ${created.operator} 接。`);
    }
  } else {
      const data = await api(`/api/threads/${current.id}/messages`, {
        method: "POST",
        token: current.token,
        body: payload,
      });
      if (data.status && activeModel() === HOME) applyStatus(data.status);
      current.updatedAt = Date.now();
    }
    remember(current);
  } catch (err) {
    painted = "";
    paint(prior);
    input.value = trimmed;
    pendingImages = images;
    showAttach();
    sentGuard.key = "";
    sending = false;
    grow();
    showSys(err.message || "消息没送出去。糖包那边暂时没应。");
    return;
  }
  for (const item of images) URL.revokeObjectURL(item.url);
  try {
    await refresh();
  } catch (err) {
    showSys(err.message || "消息送到了，这一页还没刷新。");
  } finally {
    sending = false;
    grow();
  }
}

  const button = event.target.closest(".copy");
  if (!button) return;
  const text = button.closest(".b-body").querySelector(".b-text").textContent;
  try {
    await navigator.clipboard.writeText(text);
  } catch {
    return;
  }
  button.dataset.tip = "已复制";
  button.querySelector("use").setAttribute("href", "#i-check");
  setTimeout(() => {
    button.dataset.tip = "复制";
    button.querySelector("use").setAttribute("href", "#i-copy");
  }, 1600);
});

$("#file").addEventListener("change", async () => {
  const files = [...$("#file").files];
  $("#file").value = "";
  if (!files.length) return;
  try {
    await addImages(files);
  } catch (err) {

  }

composer.addEventListener("paste", async (event) => {
  const files = [...event.clipboardData.items]
    .filter((item) => item.type.startsWith("image/"))
    .map((item) => item.getAsFile())
    .filter(Boolean);
  if (!files.length) return;
  event.preventDefault();
  try {
    await addImages(files);
  } catch (err) {
    showSys(err.message || "这张图糖包打不开。换 jpg、png、webp 或 gif。");
  }
});

function withOperator(payload) {
  if (payload instanceof FormData) {
    payload.set("operator", activeModel());
    return payload;
  }
  return { ...payload, operator: activeModel() };
}

function activeModel() {
  return threadModel || chosen;
}

function paintModels() {
  const id = activeModel();
  $("#model-name").textContent = id || "自动匹配";
  const list = $("#model-list");
  list.replaceChildren();

  const auto = document.createElement("button");
  auto.type = "button";
  auto.className = "model-row";
  auto.setAttribute("aria-pressed", String(!id));
  auto.innerHTML =
    '<i data-on="1"></i><span class="who"><b>自动匹配</b><small>按在线状态挑一个</small></span><small class="model-state">自动</small>';
  if (!id) auto.insertAdjacentHTML("beforeend", '<svg class="i"><use href="#i-check"/></svg>');
  auto.addEventListener("click", () => {
    if (current) return;
    chosen = AUTO;
    paintModels();
  });
  list.append(auto);

  for (const model of models) {
    const online = model.status === "online" || model.online === true;
    const state = model.status || (online ? "online" : "away");
    const button = document.createElement("button");
    button.type = "button";
    button.className = "model-row";
    button.disabled = !online;
    button.setAttribute("aria-disabled", String(!online));
    button.setAttribute("aria-pressed", String(model.id === id));
    button.innerHTML =
      `<i data-on="${online ? "1" : "0"}"></i><span class="who"><b></b><small></small></span><small class="model-state"></small>`;
    button.querySelector("b").textContent = model.id;
    button.querySelector("small").textContent = model.blurb || "";
    button.querySelector(".model-state").textContent =
      state === "online" ? "在线" : state === "busy" ? "忙碌" : "离开";
    if (model.id === id) button.insertAdjacentHTML("beforeend", '<svg class="i"><use href="#i-check"/></svg>');
    button.addEventListener("click", () => {
      if (current || !online) return;
      chosen = model.id;
      paintModels();
    });
    list.append(button);
  }
  if (!current) applyModelStatus();
}

async function loadModels(pick) {
  try {
    const data = await api("/api/models");
    models = data.models || [];
  } catch {
    return;
  }
  if (pick && !current) {
    let picked = "";
    try {
      picked = localStorage.getItem(MODEL_KEY) || "";
      if (picked) localStorage.removeItem(MODEL_KEY);
    } catch {}
    const fromAbout = models.find((item) => item.id === picked && item.status === "online");
    chosen = fromAbout ? fromAbout.id : AUTO;
  }
  if (!current && chosen) {
    const selected = models.find((item) => item.id === chosen);
    if (!selected || selected.status !== "online") chosen = AUTO;
  }
  paintModels();
}
